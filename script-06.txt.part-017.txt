  function finish() {
    clearTimeout(timer);
    if (st.test) { finishTest(); return; }
    const win = st.result === 'win';
    if (run) B.endRoom(run, st);
    const aliveN = st.heroes.filter(function (h) { return h.hp > 0; }).length;
    const meter = function (title, key) {
      const rows = st.heroes.slice().sort(function (a, b) { return b.stats[key] - a.stats[key]; }).filter(function (h) { return h.stats[key] > 0; });
      if (!rows.length) return '';
      const top = rows[0].stats[key];
      return '<div class="meter"><h3>' + title + '</h3>' + rows.map(function (h) {
        return '<div class="mrow" data-bd="' + h.id + '" data-k="' + key + '" role="button" tabindex="0" aria-expanded="false" style="--cc: var(--c-' + h.cls + ')"><div class="mfill" style="width: ' + (h.stats[key] / top * 100) + '%"></div><div class="mtxt"><span>' + unitBadge(h) + ' ' + esc(h.name) + '</span><span>' + Math.round(h.stats[key]) + '</span></div></div><div class="bd" hidden>' + bdHtml(h.id, key) + '</div>';
      }).join('') + '</div>';
    };
    const roles = { tank: 0, heal: 0 };
    st.heroes.forEach(function (h) { if (roles[h.role] !== undefined) roles[h.role]++; });
    const oomAny = st.heroes.some(function (h) { return h.maxMana && h.mana < h.maxMana * 0.1; });
    let tip = '';
    if (!win && st.enrageStacks > 0) tip = 'Бой затянулся, и враги вошли в ярость: не хватило урона' + (oomAny ? ', а у лекарей кончилась мана' : '') + '. Попробуй взять ещё одного бойца вместо второго лекаря или танка.';
    else if (!win && !roles.tank) tip = 'Без танка враги бьют самых опасных героев. Поставь кого-то танком: воина, паладина или друида.';
    else if (!win && !roles.heal) tip = 'Без лекаря отряд не восполняет урон. Поставь кого-то лекарем: жреца, паладина или друида.';
    else if (!win) tip = 'Попробуй другой состав: против сильных ударов хорошо работает «Аура благочестия» паладина.';
    const xp = B.xpFor(S.level, B.ENCOUNTERS[st.encounter]);
    const qDone = win && !S.amb && (!run || run.state === 'cleared') && qState(S.enc) === 1 && recordQuestVictory(S.enc,st);
    if (qDone) setQ(S.enc, 2);
    let dropTxt = '';
    QUESTS.forEach(function (dq) { if (qDone && dq.drop === S.enc && qState(dq.k) === 0) { setQ(dq.k, 2); dropTxt = 'С гнолла выпал график сбора золота: квест «' + encOf(dq.k).name + '» взят, его сдают в Златоземье.'; } });
    let ambTxt = '';
    if (S.amb && win) { const ae = B.ENCOUNTERS.ambush, acu = Math.round(20 * (1 + 0.05 * (ae.level - 1)) * (ae.cnt || 2) / 2), ar = giveXp(ae); S.money += acu; syncLevel(); save(); ambTxt = 'Засада отбита: +' + ar.main + ' опыта, +' + moneyTxt(acu) + '. Путь свободен.'; }
    else if (S.amb) ambTxt = 'Засада не отбита: ничего не потеряно, попробуй ещё раз.';
    $('now').className = 'banner';
    $('now').innerHTML = '<span class="ic" aria-hidden="true">' + (win ? '🏆' : '💀') + '</span><span><b>' + (win ? 'Победа' : 'Поражение') + '</b><br><span class="l2">Итоги на экране</span></span>';
    $('skillbar').innerHTML = '';
    $('result').innerHTML = '<h2 class="' + (win ? 'win' : 'lose') + '">' + (win ? 'Победа' : 'Поражение') + '</h2>' +
      '<div>Ходов: ' + st.actions + '. В живых: ' + aliveN + ' из ' + st.heroes.length + '.' + (win && !S.amb ? ' <b>' + (xp ? 'Опыт: ' + xp : 'Опыта нет: задание серое') + '</b>' : '') + '</div>' +
      (ambTxt ? '<p class="hint">' + esc(ambTxt) + '</p>' : '') + (tip && !S.amb ? '<p class="hint">' + esc(tip) + '</p>' : '') + (qDone ? '<p class="hint">Задание выполнено: сдай его в журнале на карте.</p>' : '') + (dropTxt ? '<p class="hint">' + esc(dropTxt) + '</p>' : '') +
      '<div class="cols2"><div>' + meter('Урон', 'dmg') + '</div><div>' + meter('Исцеление', 'heal') + '</div></div><p class="hint">Нажми на героя: урон и лечение по умениям.</p>' +
      tailHtml();
    $('result').hidden = false;
    bindTail();
  }
