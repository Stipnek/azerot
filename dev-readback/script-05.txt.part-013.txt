  /* Автоатака: базовый удар героя перед умением, вне слотов умений, нажимать не нужно.
     У ближнего боя удар оружием, у дальнего жезлом (ванд) или луком. Числа ориентировочные, баланс потом. */
  const AUTO_MULT = 0.2;
  /* Временная компенсация общего урона после урезания автоатак: атака всех героев ×1,35 (снимется при общем проходе баланса). */
  const ATKMUL = 1.35;
  const AUTO_DEFAULT = { name: 'Удар оружием', icon: '🗡️', type: 'phys' };
  const AUTO = {
    warrior: { name: 'Удар оружием', icon: '🗡️', type: 'phys' },
    paladin: { name: 'Удар оружием', icon: '🔨', type: 'phys' },

  };
  function autoOf(cls, spec) { const c = CLASSES[cls], sp = c && c.specs[spec]; return (sp && sp.auto) || AUTO[cls] || AUTO_DEFAULT; }
  const HEALAUTO = { frac: 0.55, below: 0.7, mana: 3 };
  function autoAttack(st, h) {
    if (h.hp <= 0 || !alive(st.enemies).length) return;
    if (h.role === 'heal' && (!h.maxMana || h.mana >= HEALAUTO.mana)) {
      const lw = lowestAlly(st);
      if (lw && lw.hp / lw.maxHp < HEALAUTO.below) {
        if (h.maxMana) h.mana -= HEALAUTO.mana;
        emit(st, { t: 'auto', src: h.id, name: h.cls === 'paladin' ? 'Вспышка Света' : 'Малое исцеление', icon: h.cls === 'paladin' ? '✨' : '💧' });
        heal(st, h, lw, h.healPow * HEALAUTO.frac);
        return;
      }
    }
    const a = autoOf(h.cls, h.spec), tank = h.role === 'tank', n = a.hits || 1;
    let t = tank ? tankTarget(st, h) : focusEnemy(st);
    emit(st, { t: 'auto', src: h.id, name: a.name, icon: a.icon });
    for (let i = 0; i < n; i++) {
      if (t.hp <= 0) { if (!alive(st.enemies).length) break; t = tank ? tankTarget(st, h) : focusEnemy(st); }
      damage(st, h, t, AUTO_MULT / n * (h.autoMul || 1), { type: a.type, auto: true, threatBonus: tank ? 1.3 : 1 });
    }
  }

  function use(st, h, skill) {
    if (skill === 'revenge' && !(h.revengeReady > 0)) return false;
    if (!canPayRage(h, skill)) return false;
    if (!canPayEnergy(h, skill)) return false;
    if (Object.prototype.hasOwnProperty.call(ENERGY_COST, skill)) { h.energy = Math.max(0, h.energy - ENERGY_COST[skill]); h.totalEnergySpent = (h.totalEnergySpent || 0) + ENERGY_COST[skill]; }
    if (Object.prototype.hasOwnProperty.call(RAGE_COST, skill)) {
      const paid = rageCost(h, skill);
      h.rage = Math.max(0, h.rage - paid);
      h.lastRageSpent = paid;
      if (['shieldslam','thunder','bloodthirst','whirlwind','maul','swipe'].indexOf(skill) !== -1) h.lastMainRageSkill = skill;
      h.totalRageSpent = (h.totalRageSpent || 0) + paid;
    }
    if (skill === 'revenge') h.revengeReady = 0;
    h.cd[skill] = SKILLS[skill].cd;
    if (h.arcConc && st.rnd() < 0.10) { const ks = Object.keys(h.cd).filter(function (k) { return k !== skill && h.cd[k] > 0; }); if (ks.length) { const k = ks[Math.floor(st.rnd() * ks.length)]; h.cd[k] = 0; log(st, h.name + ': «Чародейская концентрация» сбросила перезарядку «' + SKILLS[k].name + '»', 'good'); } }
    if (skill !== 'pom') { const pm = h.effects.find(function (f) { return f.kind === 'pom'; }); if (pm) { h.effects = h.effects.filter(function (f) { return f !== pm; }); h.pomNow = true; } }
    const c = SKILLS[skill].mana;
    if (c && h.maxMana) h.mana = Math.max(0, h.mana - c);
    emit(st, { t: 'skill', src: h.id, skill });
    return true;
  }

  function focusEnemy(st) { const f = alive(st.enemies); f.sort(function (a, b) { return a.hp - b.hp; }); return f[0]; }
  function tankTarget(st, h) { const f = alive(st.enemies); return f.find(function (e) { return e.aggro !== h.id; }) || focusEnemy(st); }
  function lowestAlly(st) { const a = alive(st.heroes); a.sort(function (x, y) { return pct(x) - pct(y); }); return a[0]; }
  function mostTargeted(st) {
    const counts = {};
    alive(st.enemies).forEach(function (e) { if (e.aggro) counts[e.aggro] = (counts[e.aggro] || 0) + (e.boss ? 3 : 1); });
    let best = null, bestV = 0;
    alive(st.heroes).forEach(function (h) { if ((counts[h.id] || 0) > bestV) { bestV = counts[h.id]; best = h; } });
    return best;
  }

  function doTaunt(st, h, skill, one) {
    const s = SKILLS[skill];
    (one ? [one] : alive(st.enemies)).forEach(function (e) {

      Object.keys(e.threat).forEach(function (k) { max = Math.max(max, e.threat[k]); });
      e.threat[h.id] = max * 1.1 + 50;
      e.effects = e.effects.filter(function (f) { return f.kind !== 'taunt'; });
      e.effects.push({ kind: 'taunt', target: h.id, turns: 3, name: s.name + ': ' + h.name, icon: s.icon });
    });
    log(st, h.name + ': «' + s.name + '», все враги переключаются на него', 'skill');
  }
