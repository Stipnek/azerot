  function float(id, text, cls) {
    const t = toks[id]; if (!t) return;
    const c = center(t.tc);
    const s = document.createElement('span');
    s.textContent = text; s.className = 'ffct ' + (cls || '');
    s.style.left = (c.x + (Math.random() * 30 - 15)) + 'px'; s.style.top = (c.y - 6) + 'px';
    $('fx').appendChild(s);
    setTimeout(function () { s.remove(); }, 1050);
  }
  function glow(id, col, ms) {
    const t = toks[id]; if (!t || reduced) return;
    const tc = t.tc;
    if (tc.classList.contains('spr')) {
      const pz = tc.dataset.pose || 'idle';
      tc.querySelectorAll(pz === 'idle' ? '.spi.idle, .spi.idle2' : '.spi.' + pz).forEach(function (im) { im.animate([{ filter: 'drop-shadow(0 0 3px ' + col + ') drop-shadow(0 0 10px ' + col + ')' }, { filter: 'drop-shadow(0 0 0 transparent)' }], { duration: ms }); });
    } else {
      tc.animate([{ boxShadow: '0 0 0 3px ' + col + ', 0 0 18px 4px ' + col }, { boxShadow: '0 5px 0 rgba(0,0,0,.35)' }], { duration: ms });
    }
  }
  function hitFx(e) {
    if (!st) return;
    const tgt = B.byId(st, e.tgt); if (!tgt || !toks[e.tgt]) return;
    if (e.t === 'dmg') {
      const shown = e.miss ? 'Промах' : e.dodge ? 'Уклонение' : (e.amt ? ((tgt.side === 'hero' ? '−' : '') + Math.round(e.amt) + (e.absorbed ? ' (щит ' + Math.round(e.absorbed) + ')' : '')) : ('щит ' + Math.round(e.absorbed)));
      float(e.tgt, shown, (tgt.side === 'hero' && e.amt ? 'hurt' : '') + (e.crit ? ' crit' : '') + (e.dot ? ' tick' : ''));
      if (!e.dot && e.amt) setPose(e.tgt, 'hit', 340);
      if (!reduced && !e.dot && !e.dodge && !e.miss) toks[e.tgt].tc.animate([{ transform: 'translateX(0)', filter: 'brightness(1)' }, { transform: 'translateX(-5px)', filter: 'brightness(2) saturate(.3)' }, { transform: 'translateX(5px)' }, { transform: 'translateX(0)', filter: 'brightness(1)' }], { duration: 260 });
    } else if (e.t === 'heal' && e.amt > 0) {
      float(e.tgt, '+' + Math.round(e.amt), 'heal' + (e.crit ? ' crit' : '') + (e.hot ? ' tick' : ''));
      if (!e.hot) glow(e.tgt, '#6dff7a', 500);
    } else if ((e.t === 'buff' || e.t === 'debuff') && !reduced) {
      const col = e.t === 'buff' ? '#6aa6ff' : '#b46bff';
      glow(e.tgt, col, 550);
    }
  }
  function projectile(fromId, toId, icon, dur) {
    const a = center(toks[fromId].tc), b = center(toks[toId].tc);
    const p = document.createElement('span');
    p.className = 'proj'; if (String(icon).indexOf('<span class="az-skill-icon"') === 0) p.innerHTML = icon; else p.textContent = icon; p.style.left = a.x + 'px'; p.style.top = a.y + 'px';
    $('fx').appendChild(p);
    const dx = b.x - a.x, dy = b.y - a.y;
    p.animate([
      { transform: 'translate(-50%, -50%) scale(.7)' },
      { transform: 'translate(calc(-50% + ' + dx / 2 + 'px), calc(-50% + ' + (dy / 2 - 24) + 'px)) scale(1.15)' },
      { transform: 'translate(calc(-50% + ' + dx + 'px), calc(-50% + ' + dy + 'px)) scale(.9)' }
    ], { duration: dur, easing: 'ease-in' });
    setTimeout(function () { p.remove(); }, dur);
  }

  }

    if (actIdx === -1) return;
    const after = evs.slice(actIdx + 1);
    const actor = B.byId(st, evs[actIdx].id);
    if (!actor || !toks[actor.id]) return;
    if (actor.ghost) { after.forEach(function (e) { if (e.t === 'dmg' || e.t === 'heal') hitFx(e); }); return; }

    const effects = after.filter(function (e) { return e.t === 'dmg' || (e.t === 'heal' && e.amt > 0) || e.t === 'buff' || e.t === 'debuff'; });
    const D = Math.min(760, 600 / speed * 0.72);
