  function dungeonXp(level, key) { return DUNGEONS[key].rooms.reduce(function (a, r) { return a + xpFor(level, ENCOUNTERS[r]); }, 0); }
  function newRun(dk, party, main, level, seed) {
    const D = DUNGEONS[dk];
    return { dk: dk, D: D, party: party, main: main, level: level, rooms: D.rooms.slice(), idx: 0, rests: D.rests, restsMax: D.rests, carry: null, state: 'room', loot: [], xp: 0, kills: 0, seed: seed || 1, rnd: makeRng((seed || 1) + 99), usedRest: 0 };
  }
  function roomKey(run) { return run.rooms[run.idx]; }
  function startRoom(run, seed) {
    return createBattle({ party: run.party, main: run.main, encounter: roomKey(run), level: run.level, seed: seed || ((run.seed + run.idx * 7919) >>> 0), carry: run.carry });
  }
  function endRoom(run, st) {
    const key = roomKey(run);
    if (st.result === 'win') {
      run.kills++;
      run.xp += xpFor(run.level, ENCOUNTERS[key]);
      const table = LOOT[key] || [];
      if (table.length) run.loot.push({ room: key, item: table[Math.floor(run.rnd() * table.length)] });
      const rc = RECOVERY;
      run.carry = st.heroes.map(function (h) {
        if (h.hp <= 0) return { hp: 0, maxHp: h.maxHp, mana: h.mana, maxMana: h.maxMana };
        return { hp: Math.min(h.maxHp, h.hp + h.maxHp * rc.hp), maxHp: h.maxHp, mana: h.maxMana ? Math.min(h.maxMana, h.mana + h.maxMana * rc.mana) : 0, maxMana: h.maxMana };
      });
      run.idx++;
      run.state = run.idx >= run.rooms.length ? 'cleared' : 'between';
    } else {
      run.state = 'wipe';
    }
  }
  function runStatus(run) {
    const c = run.carry || [];
    if (!c.length) return { avgHp: 1, dead: 0, minMana: null };
    const avgHp = c.reduce(function (a, x) { return a + x.hp / x.maxHp; }, 0) / c.length;
    const mana = c.filter(function (x) { return x.maxMana > 0; }).map(function (x) { return x.mana / x.maxMana; });
    return { avgHp: avgHp, dead: c.filter(function (x) { return x.hp <= 0; }).length, minMana: mana.length ? Math.min.apply(null, mana) : null };
  }
  function restRun(run) {
    if (run.rests <= 0 || !run.carry) return false;
    run.rests--; run.usedRest++;
    run.carry = run.carry.map(function (x) { return { hp: x.maxHp, maxHp: x.maxHp, mana: x.maxMana, maxMana: x.maxMana }; });
    return true;
  }
  function autoDecide(run) {
    const s = runStatus(run);
    const weak = s.avgHp < RESTAT.hp || s.dead > 0 || (s.minMana !== null && s.minMana < RESTAT.mana);
    if (!weak) return 'next';
    if (run.rests > 0) return 'rest';
    return (s.avgHp < 0.35 || s.dead >= 2) ? 'exit' : 'next';
  }
  function simulateRun(run) {
    const log = [];
    let guard = 0;
    while ((run.state === 'room' || run.state === 'between') && guard++ < 40) {
      if (run.state === 'between') {
        const d = autoDecide(run);
        log.push(d);
        if (d === 'rest') restRun(run);
        else if (d === 'exit') { run.state = 'exit'; break; }
        run.state = 'room';
        continue;
      }
      const st = startRoom(run);
      while (!st.over) step(st);
      endRoom(run, st);
    }
    return { state: run.state, kills: run.kills, rooms: run.rooms.length, xp: run.xp, loot: run.loot.slice(), rests: run.usedRest, decisions: log };
  }
