  function canPayRage(h, skill) {
    if (!Object.prototype.hasOwnProperty.call(RAGE_COST, skill)) return true;
    return !!h.maxRage && Number.isFinite(h.rage) && h.rage + 1e-9 >= RAGE_COST[skill];
  }
  function rageCost(h, skill) {
    if (!canPayRage(h, skill)) return null;
    if (skill === 'execute') return h.rage;
    if (skill === 'regen') return Math.min(30, h.rage);
    return RAGE_COST[skill] || 0;
  }
