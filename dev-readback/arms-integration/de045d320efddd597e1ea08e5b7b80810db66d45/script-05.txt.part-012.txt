    if (tgt.hp <= 0) kill(st, tgt);
    else if (src.cls === 'rogue' && tgt.side === 'enemy' && !opt.dot) breakArmor(src, tgt);
    return dealt + absorbed;

  function heal(st, src, tgt, raw, opt) {
    opt = opt || {};
    if (tgt.hp <= 0) return 0;
    let amt = raw * (0.9 + st.rnd() * 0.2);
    const fav = src.effects && !opt.hot ? src.effects.find(function (f) { return f.kind === 'favor'; }) : null;
    const crit = fav ? true : (!opt.noCrit && st.rnd() < (src.crit || 0));
    if (fav) src.effects = src.effects.filter(function (f) { return f !== fav; });
    if (crit) amt *= 1.5;
    if (src.effects && src.effects.some(function (f) { return f.kind === 'pi'; })) amt *= 1.2;
    if (src.raceAll) amt *= 1 + src.raceAll;
    if (crit && src.pids) { if (src.pids.indexOf('illum') !== -1 && src.maxMana) src.mana = Math.min(src.maxMana, src.mana + 5); if (src.pids.indexOf('inspir') !== -1) addEffect(tgt, { kind: 'armorup', amt: 0.20, turns: 2, src: src.id, name: 'Вдохновение', icon: '✨' }); }
    const hr = tgt.effects.reduce(function (a, f) { return f.kind === 'healred' ? a + f.amt * (f.stacks || 1) : a; }, 0);
    if (hr) amt *= Math.max(0.1, 1 - hr);
    const hu = tgt.effects.reduce(function (a, f) { return f.kind === 'healup' ? a + f.amt : a; }, 0);
    if (hu) amt *= 1 + hu;
    amt = Math.round(amt);
    const eff = Math.min(amt, tgt.maxHp - tgt.hp);
    tgt.hp += eff;
    if (st.test && src.side === 'hero' && !src.dummy) st.hlog.push([src.stats.turns || 0, src.id, eff, amt - eff]);
    if (src.side === 'hero' && eff > 0) {
      src.stats.heal += eff;
      const foes = alive(st.enemies);
      foes.forEach(function (e) { addThreat(e, src, eff * 0.5 / Math.max(1, foes.length)); });
    }
    emit(st, { t: 'heal', src: src.id, tgt: tgt.id, amt: eff, crit, hot: !!opt.hot });
    return eff;
  }

  function kill(st, u) {
    u.hp = 0; u.effects = [];
    emit(st, { t: 'death', id: u.id });
    log(st, u.name + ' погибает', u.side === 'hero' ? 'bad' : 'good');
    if (u.side === 'enemy' && u.onDeath && !u.flags.spawned) {
      u.flags.spawned = true;
      const ne = spawnEnemy(st, u.onDeath);
      log(st, 'Из Крошилки выбирается ' + ne.name, 'yell');
      emit(st, { t: 'yell', src: ne.id });
    }
    if (u.side === 'hero') st.enemies.forEach(function (e) { delete e.threat[u.id]; });
  }

  function addEffect(u, eff) {
    u.effects = u.effects.filter(function (f) { return !(f.kind === eff.kind && f.src === eff.src && f.name === eff.name); });
    u.effects.push(eff);
  }

  /* Неистовство: скорость растёт ступенями по 10% потерянного ХП, максимум +60% при ХП ниже 10%; лечение возвращает скорость. */
  const FURY = { step: 0.0667, max: 9 };
  const DEATHWISH = { dealt: 1.20, taken: 1.05 };
  function effSpd(u) {
    let m = 1;
    if (u.fury) m += Math.min(FURY.max, Math.max(0, Math.floor((1 - u.hp / u.maxHp) * 10))) * FURY.step;
    const hs = u.effects ? u.effects.find(function (f) { return f.kind === 'haste'; }) : null;
    if (hs) m += hs.amt;
    const sl = u.effects ? u.effects.find(function (f) { return f.kind === 'slow'; }) : null;
    if (sl) m -= sl.amt;
    const ab = u.effects ? u.effects.find(function (f) { return f.kind === 'arcb'; }) : null;
    if (ab) m -= 0.05 * ab.stacks;
    return u.spd * m;
  }
  function nextActor(st) {
    const units = all(st).filter(function (u) { return u.hp > 0; });
    let ready = units.filter(function (u) { return u.tm >= TM; });
    if (!ready.length) {
      let ticks = Infinity;
      units.forEach(function (u) { ticks = Math.min(ticks, Math.ceil((TM - u.tm) / effSpd(u))); });
      units.forEach(function (u) { u.tm += effSpd(u) * ticks; });
      ready = units.filter(function (u) { return u.tm >= TM; });
    }
    ready.sort(function (a, b) { return b.tm - a.tm; });
    return ready[0];
  }
