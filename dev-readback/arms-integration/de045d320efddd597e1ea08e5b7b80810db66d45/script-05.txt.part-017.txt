  function enemyTarget(st, e) {
    refreshAggro(st, e);
    let t = e.aggro ? byId(st, e.aggro) : null;
    if (!t || t.hp <= 0) {
      const a = targetsOf(st);
      t = a[Math.floor(st.rnd() * a.length)];
      if (t) { e.threat[t.id] = (e.threat[t.id] || 0) + 1; e.aggro = t.id; }
    }
    return t;
  }

  function eskill(st, e, id, cd) {
    e.cd[id] = cd;
    const s = e.eskills.find(function (x) { return x.id === id; });
    emit(st, { t: 'eskill', src: e.id, id, name: s ? s.name : id, icon: s ? s.icon : '' });
  }

    if (!heroes.length) return;

        const t = enemyTarget(st, e);
        if (ready('smash')) { eskill(st, e, 'smash', 3); damage(st, e, t, 1.7); return; }
        damage(st, e, t, 1.0); return;
      }
