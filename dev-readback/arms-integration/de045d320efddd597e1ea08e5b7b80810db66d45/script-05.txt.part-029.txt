  function armsValidate(alloc, level, opt) {
    opt = opt || {};
    const errors = [], a = alloc && typeof alloc === 'object' ? alloc : {}, perTier = [0, 0, 0, 0, 0, 0, 0, 0];
    let total = 0;
    Object.keys(a).forEach(function (id) {
      const r = a[id], n = ARMS_NODE[id];
      if (!n) { errors.push('Неизвестный узел: ' + id); return; }
      if (!Number.isInteger(r) || r < 0 || r > n.max) { errors.push(n.ru + ': недопустимый ранг ' + r); return; }
      if (r > 0 && !opt.ignoreAvailability && ARMS_PROTO.unavailable.indexOf(id) !== -1) errors.push(n.ru + ': недоступно в прототипе');
      total += r; perTier[n.tier] += r;
    });
    const budget = armsPoints(level);
    if (total > budget) errors.push('Вложено ' + total + ' при бюджете ' + budget);
    for (let t = 2; t <= 7; t++) {
      if (!perTier[t]) continue;
      let below = 0; for (let k = 1; k < t; k++) below += perTier[k];
      if (below < 5 * (t - 1)) errors.push('Ряд ' + t + ' требует ' + 5 * (t - 1) + ' очков в нижних рядах, есть ' + below);
    }
    ARMS_TREE.forEach(function (n) { if (n.req && (a[n.id] || 0) > 0 && (a[n.req] || 0) < ARMS_NODE[n.req].max) errors.push(n.ru + ' требует полного узла «' + ARMS_NODE[n.req].ru + '»'); });
    return { ok: errors.length === 0, errors: errors, spent: total, budget: budget };
  }
  function armsChange(alloc, id, delta, level) {
    const next = Object.assign({}, alloc || {}); next[id] = (next[id] || 0) + delta; if (!next[id]) delete next[id];
    const v = armsValidate(next, level);
    return v.ok ? { ok: true, alloc: next, errors: [] } : { ok: false, alloc: Object.assign({}, alloc || {}), errors: v.errors };
  }

  function protoRageNeed(h, skill) {
