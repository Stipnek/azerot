  function updateBoss() {
    const boss = st.enemies.find(function (e) { return e.boss && e.hp > 0; }) || st.enemies.find(function (e) { return e.boss; });
    const el = $('bossbar');

    $('bb-i').style.width = Math.max(0, boss.hp / boss.maxHp * 100) + '%';

  function drawAggro() {
    const svg = $('aggro'), f = stageRect();
    svg.setAttribute('viewBox', '0 0 ' + f.width + ' ' + f.height);
    let out = '<defs><marker id="ah" viewBox="0 0 10 10" refX="8" refY="5" markerWidth="6" markerHeight="6" orient="auto"><path d="M1 1L9 5L1 9z" fill="#ff5a4a"/></marker></defs>';
    if (!st.over) st.enemies.forEach(function (e) {
      if (e.ghost || e.hp <= 0 || !e.aggro || !toks[e.id] || !toks[e.aggro]) return;
      const tg = B.byId(st, e.aggro); if (!tg || tg.hp <= 0) return;
      const a = topPoint(toks[e.id].el), b = topPoint(toks[e.aggro].el);
      const lift = Math.max(24, f.height * 0.2 + Math.abs(a.x - b.x) * 0.08);
      const cx = (a.x + b.x) / 2, cy = Math.min(a.y, b.y) - lift;
      const taunted = e.effects.some(function (x) { return x.kind === 'taunt'; });
      out += '<path d="M' + a.x + ' ' + a.y + ' Q' + cx + ' ' + cy + ' ' + b.x + ' ' + b.y + '" fill="none" stroke="#ff5a4a" stroke-width="' + (taunted ? 3 : 2) + '" stroke-dasharray="' + (taunted ? '0' : '6 5') + '" stroke-opacity=".9" marker-end="url(#ah)"/>';
    });
    svg.innerHTML = out;
  }
