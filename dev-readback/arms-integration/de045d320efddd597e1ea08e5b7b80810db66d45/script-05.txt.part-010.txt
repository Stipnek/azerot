tTargets: opts.party.length === 1 };
    opts.party.forEach(function (p) {
      const c = CLASSES[p.cls];
      const specKey = c.specs[p.spec] ? p.spec : c.def;
      const s = c.specs[specKey];
      const hl = p.level || lvl, f = lvlFactor(hl), hs = heroStats(p.cls, specKey, hl, p.statBonuses);
      /* Ролевой баланс: танк держит удар, но не конкурирует с ДД по урону; хил платит за лечение личным DPS. */
      const roleHp = s.role === 'tank' ? 1.08 : s.role === 'heal' ? 1.04 : 0.98;
      const roleAtk = s.role === 'tank' ? 0.90 : s.role === 'heal' ? 0.85 : 1.04;
      const roleHeal = s.role === 'heal' ? 1.08 : 1;
      const rb = p.rb || 0;
      const maxHp = Math.round((s.hp * f + Math.max(0,Number(p.statBonuses && p.statBonuses.stamina)||0)*10) * roleHp * (1 + (aura.hp || 0)) * (1 + rb + protoAcc(opts, p)));
      const soloHealer = opts.party.length === 1 && s.role === 'heal';

        id: 'h' + (st.uid++), side: 'hero', level: hl, cls: p.cls, spec: specKey, ai: p.cls + '_' + specKey, name: p.nm || c.name, specName: s.name, icon: c.icon, role: s.role,

        skills: s.skills.slice(), cd: {}, effects: [], tm: rnd() * 150, isMain: p.main !== undefined ? !!p.main : p.cls === mainCls,

ENERGY_CFG.start : 0, maxRage: hs.resource === 'rage' ? RAGE_CFG.max : 0, rage: RAGE_CFG.start, rageTakenThisTurn: 0, maxMana: Math.round(hs.mana * (1 + rb)), mana: Math.round(hs.mana * (1 + rb)), oom: false, oomHit: false, maxCp: s.combo || 0, cp: 0,

      });
    });
    opts.party.forEach(function (p, i) {
      const rc = RACES[p.race]; if (!rc) return; const h = st.heroes[i];
      h.race = p.race; h.racial = rc.active || null; h.racePassive = rc.passive;
      if (p.race === 'human') h.raceAll = 0.03;
      if (p.race === 'dwarf') h.res += 0.02;
      if (p.race === 'gnome') { h.primaryMultiplier *= 1.02; if (h.role === 'heal') h.healPow *= 1.02; else h.atk *= 1.02; } /* +2% к главной характеристике: лекарю к лечению, остальным к атаке */
      if (p.race === 'nelf') h.spd *= 1.02;
    });
    st.heroes.forEach(function (h) { const sq = CLASSES[h.cls] && CLASSES[h.cls].specs[h.spec]; if (sq && sq.stealth) h.effects.push({ kind: 'hidden', turns: 999, src: h.id, name: 'Скрытность', icon: '🥷' }); });
    st.heroes.forEach(function (h) {
      const sp = CLASSES[h.cls] && CLASSES[h.cls].specs[h.spec]; if (!sp || !sp.passives) return;
      h.pids = sp.passives.map(function (x) { return x.id; });
      sp.passives.forEach(function (p) { if (p.id === 'dualwield') h.autoMul = 1.25; if (p.id === 'deflect') { if (h.ai === 'warrior_prot') h.parryBase = 0.05; else h.dodgeBase = 0.05; } if (p.id === 'thickhide') h.armor *= 1.10; if (p.id === 'giftnature') h.healPow *= 1.10; if (p.id === 'impwps') h.impwps = true; if (p.id === 'shatter') h.shatter = true; if (p.id === 'ruin') h.critMul = 2.0; if (p.id === 'mortal') h.critMul = 1.65; if (p.id === 'arcconc') h.arcConc = true; if (p.id === 'arcinst') { h.crit += 0.03; h.mdAll = (h.mdAll || 0) + 0.03; } });
    });
    if (opts.experiment || opts.party.some(function (p) { return p.proto; })) protoSetupAll(st, opts);
    opts.party.forEach(function (p, i) {
      if (!p.pet) return;
      const c = CLASSES[p.cls], sk = c.specs[p.spec] ? p.spec : c.def;
      const pet = createPet(st, st.heroes[i], p.pet, c.specs[sk], lvlFactor(p.level || lvl), aura), ps = c.specs[sk].passives, o = st.heroes[i];
      if (pet && ps) {
        if (ps.some(function (x) { return x.id === 'unholy'; })) pet.atk *= 1.2;
        if (ps.some(function (x) { return x.id === 'frenzy'; })) pet.frenzy = true;
        if (ps.some(function (x) { return x.id === 'masterdemo'; })) { if (pet.spec === 'imp') o.mdMagic = 0.10; if (pet.spec === 'succubus') o.mdAll = 0.05; if (pet.spec === 'voidwalker') o.mdTaken = 0.10; if (pet.spec === 'felhunter') o.crit += 0.10; }
      }
    });
    if (opts.carry) st.heroes.forEach(function (h, i) {
      const c = opts.carry[i]; if (!c) return;
      h.hp = Math.max(0, Math.min(h.maxHp, c.hp));
      if (h.maxMana) h.mana = Math.max(0, Math.min(h.maxMana, Number.isFinite(c.mana) ? c.mana : h.maxMana));
    });
    ENCOUNTERS[opts.encounter].enemies.forEach(function (key) { spawnEnemy(st, key); });
    if (opts.test) setupTest(st, opts.test);
    st.heroes.forEach(function (h) { if (h.role === 'tank' && h.hp > 0) st.enemies.forEach(function (e) { e.threat[h.id] = 40; }); });
    st.enemies.forEach(function (e) { refreshAggro(st, e); });
    return st;
  }

  function directHit(st, src, tgt, amt) {
    amt = Math.max(1, Math.round(amt)); tgt.hp -= amt; tgt.stats.taken += amt;
    emit(st, { t: 'dmg', src: src.id, tgt: tgt.id, amt: amt, absorbed: 0, crit: false, dot: false, auto: false });
    if (tgt.hp <= 0) { st.dummyDeaths++; tgt.hp = tgt.maxHp; tgt.effects = []; log(st, tgt.name + ' погибает и сразу воскресает', 'info'); }
  }
  function spawnEnemy(st, key) {
    const d = ENEMIES[key];
    const elvl = ENCOUNTERS[st.encounter].level, ef = lvlFactor(elvl), gsc = groupScale(st);

      id: 'e' + (st.uid++), side: 'enemy', key, name: d.name, icon: d.icon, boss: !!d.boss,
      maxHp: Math.round(d.hp * ef * gsc * 1.08), hp: Math.round(d.hp * ef * gsc * 1.08), atk: d.atk * ef * gsc * 1.06, armor: d.armor, res: d.res, spd: d.spd, crit: 0.05, ai: d.ai, level: elvl,

      cd: {}, effects: [], tm: st.rnd() * 150, threat: {}, aggro: null, flags: {},
    };
    if(key==='v56_morbent' && st.morbentsBane){e.armor=.12;e.res=.05;}
    st.enemies.push(e);
    return e;
  }
