  function enrageIn(st) {
    if (!st.enrage) return null;
    if (st.actions < st.enrage.after) return st.enrage.after - st.actions;
    return st.enrage.every - ((st.actions - st.enrage.after) % st.enrage.every);
  }
  function enrageTick(st) {
    const c = st.enrage; if (!c) return;
    if (st.actions >= c.after && (st.actions - c.after) % c.every === 0) {
      st.enrageStacks++;
      const pct = Math.round(st.enrageStacks * c.step * 100);
      log(st, st.enrageStacks === 1 ? 'Бой затянулся: враги приходят в ярость, их урон растёт каждые ' + c.every + ' ходов (сейчас +' + pct + '%)' : 'Ярость врагов растёт: урон +' + pct + '%', 'yell');
      emit(st, { t: 'enrage', stacks: st.enrageStacks, pct: pct });
    }
  }

  function step(st) {
    st.events = [];
    if (st.over) return st.events;
    const u = nextActor(st);
    if (!u) return st.events;
    u.tm -= TM;
    st.actions++;
    st.current = u.id;
    emit(st, { t: 'turn', id: u.id });
    enrageTick(st);
    startTurn(st, u);
    if (u.side === 'hero' && !u.pet) petsTick(st, u);
    if (u.hp > 0 && !checkEnd(st)) { emit(st, { t: 'act', id: u.id }); if (st.test && u.side === 'hero' && !u.pet && !u.dummy) { u.stats.turns = (u.stats.turns || 0) + 1; if (u.isMain || (u === st.heroes[0] && !st.heroes.some(function (x) { return x.isMain; }))) st.mainTurns++; }
    if (u.pet) petAct(st, u); else if (u.dummy) { emit(st, { t: 'idle', id: u.id }); } else if (u.side === 'hero') { racialAct(st, u); if (u.effects.some(function (f) { return f.kind === 'stun'; })) { emit(st, { t: 'eskill', src: u.id, id: 'stun', name: 'Оглушён', icon: '💫' }); log(st, u.name + ' оглушён и пропускает ход', 'info'); } else if (u.role !== 'heal' && !autoOf(u.cls, u.spec).ranged && u.effects.some(function (f) { return f.kind === 'root'; })) { emit(st, { t: 'eskill', src: u.id, id: 'root', name: 'Обездвижен', icon: '🕸️' }); log(st, u.name + ' обездвижен и не может подойти к врагу', 'info'); } else { if (!u.effects.some(function (f) { return f.kind === 'hidden'; })) { autoAttack(st, u); if (u.extraAuto) { u.extraAuto = false; autoAttack(st, u); } } heroAct(st, u); u.pomNow = false; } } else if (u.effects.some(function (f) { return f.kind === 'stun'; })) { emit(st, { t: 'eskill', src: u.id, id: 'stun', name: 'Оглушён', icon: '💫' }); log(st, u.name + ' оглушён и пропускает ход', 'info'); } else enemyAct(st, u); }
    if (u.revengeReady > 0) u.revengeReady--;
    if (u.side === 'hero' && u.oomHit) {
      u.oomHit = false;
      if (!u.oom) { u.oom = true; emit(st, { t: 'oom', id: u.id }); log(st, u.name + ': не хватает маны для одного из умений', 'bad'); }
    }
    (st.pets || []).forEach(function (p) { const o = byId(st, p.owner); if (o && o.hp <= 0 && !p.gone) { p.hp = 0; p.gone = true; } });
    st.enemies.forEach(function (e) { if (e.hp > 0) refreshAggro(st, e); });
    checkEnd(st);
    if (!st.over && st.actions >= MAX_ACTIONS) { st.over = true; st.result = 'lose'; log(st, 'Бой затянулся. Отряд отступает', 'bad'); }
    return st.events;
  }
