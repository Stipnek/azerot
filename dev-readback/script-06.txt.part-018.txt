  function runSummaryHtml() {
    if (run.state === 'cleared') setCleared(run.dk);
    const total = run.rooms.length;
    const loot = run.loot.length ? '<ul>' + run.loot.map(function (l) { return '<li>' + esc(B.ENCOUNTERS[l.room].name) + ': ' + esc(l.item) + '</li>'; }).join('') + '</ul>' : ' нет.';
    return '<div class="runsum"><b>' + (run.state === 'cleared' ? 'Данж пройден' : (run.state === 'exit' ? 'Вы вышли с лутом' : 'Забег окончен')) + '</b><br>Боссов убито: ' + run.kills + ' из ' + total + '. Опыт: ' + run.xp + '. Отдыхов использовано: ' + run.usedRest + '.<br>Добыча:' + loot + '</div>' +
      '<div class="btns">' + (run.state === 'cleared' ? '' : '<button class="btn" id="again">Ещё раз</button>') + '<button class="btn" id="back">На карту</button></div>';
  }
  function tailHtml() {
    if (run && run.state === 'between') {
      return '<p class="hint" id="runstat">' + statusLine() + ' ' + lootLine() + '</p><div class="btns"><button class="btn" id="rnext">Дальше</button><button class="btn" id="rrest"' + (run.rests > 0 ? '' : ' disabled') + '>Отдых (' + run.rests + ')</button><button class="btn" id="rexit">Выйти</button></div>';
    }
    if (run && (run.state === 'cleared' || run.state === 'wipe')) return runSummaryHtml();
    return '<div class="btns">' + (st && st.result === 'win' ? '' : '<button class="btn" id="again">Ещё раз</button>') + '<button class="btn" id="back">' + (S.amb && st && st.result === 'win' ? 'Продолжить путь' : 'На карту') + '</button></div>';
  }
  function bindTail() {
    clearTimeout(autoTimer);
    const q = function (id, fn) { const el = $(id); if (el) el.addEventListener('click', fn); };
    q('again', startBattle); q('back', toPrep); q('rnext', runNext); q('rrest', runRest); q('rexit', runExit);
    if (run && run.state === 'between' && S.auto && autoAllowed()) autoStep();
  }
  function runNext() { clearTimeout(autoTimer); $('result').hidden = true; beginBattle(B.roomKey(run), run.carry); }
  function runRest() {
    clearTimeout(autoTimer);
    if (!B.restRun(run)) return;
    $('runstat').innerHTML = statusLine() + ' Отряд отдохнул.';
    $('rrest').disabled = run.rests === 0; $('rrest').textContent = 'Отдых (' + run.rests + ')';
    renderRoom();
  }
  function runExit() {
    clearTimeout(autoTimer);
    run.state = 'exit';
    $('result').innerHTML = '<h2 class="lose">Вы вышли</h2>' + runSummaryHtml();
    bindTail();
  }
  function autoStep() {
    const d = B.autoDecide(run);
    const label = d === 'rest' ? 'отдыхаем' : (d === 'exit' ? 'выходим с лутом' : 'идём дальше');
    $('runstat').innerHTML = statusLine() + ' <b>Авто: ' + label + '</b>';
    autoTimer = setTimeout(function () {
      if (d === 'rest') { B.restRun(run); $('runstat').innerHTML = statusLine() + ' <b>Авто: отряд отдохнул, идём дальше</b>'; renderRoom(); autoTimer = setTimeout(runNext, 1300); }
      else if (d === 'exit') runExit();
      else runNext();
    }, 1500);
  }
  function skipRun() {
    run = B.newRun(S.enc.slice(8), partyArg(), S.main, S.level, (Math.random() * 1e9) >>> 0);
    B.simulateRun(run);
    clearTimeout(timer); clearTimeout(autoTimer); st = null; lines = [];
    battleShow();
    $('tokens').innerHTML = ''; $('aggro').innerHTML = ''; $('skillbar').innerHTML = ''; $('order').innerHTML = ''; $('enr').textContent = ''; $('room').textContent = '';
    $('bossbar').classList.remove('on'); $('card').hidden = true; $('log').classList.remove('open'); $('log').innerHTML = '';
    $('scene').innerHTML = sceneSvg('deadmines');
    $('now').className = 'banner'; $('now').innerHTML = '<span class="ic" aria-hidden="true">⏩</span><span><b>Забег пропущен</b><br><span class="l2">Итог рассчитан сразу</span></span>';
    const ok = run.state === 'cleared';
    if (ok && qState(S.enc) === 1) setQ(S.enc, 2);
    $('result').innerHTML = '<h2 class="' + (ok ? 'win' : 'lose') + '">' + (ok ? 'Победа' : (run.state === 'exit' ? 'Вы вышли' : 'Поражение')) + '</h2>' + runSummaryHtml();
    $('result').hidden = false;
    bindTail();
  }

  function toPrep() {
    S.polyOn = false;
    const amb = S.amb, won = !!(st && st.result === 'win');
    clearTimeout(timer); clearTimeout(autoTimer); st = null; run = null; S.amb = null; playing = false;
    showWorld('map');
    if (amb && won && amb.resume) amb.resume();
    else if (amb) $('jmsg').textContent = 'Засада не отбита: ничего не потеряно, попробуй ещё раз.';
  }
