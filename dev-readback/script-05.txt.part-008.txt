  function leaderAura(party) {
    const a = { res: [] };
    const ps = (party || []).filter(function (p) { return p && CLASSES[p.cls]; });
    const lead = ps.find(function (p) { return p.main; }) || ps[0];
    if (!lead) return a;
    const c = CLASSES[lead.cls], specKey = c.specs[lead.spec] ? lead.spec : c.def;
    const A = AURAS[lead.cls + ':' + specKey];
    if (A) {
      ['physDmg', 'magicDmg', 'allDmg', 'hp', 'dmgTaken', 'crit', 'heal', 'spd'].forEach(function (f) { if (A.b[f]) a[f] = A.b[f]; });
      if (A.b.lowHp) a.lowHp = 1;
      a.name = A.n; a.text = A.t;
    }
    const comps = ps.filter(function (p) { return p !== lead; });
    let tank = 0, heal = 0, dd = 0, phys = 0, magic = 0, same = 0, pet = 0;
    comps.forEach(function (p) {
      const c2 = CLASSES[p.cls], s2 = c2.specs[c2.specs[p.spec] ? p.spec : c2.def];
      if (s2.role === 'tank') tank++; else if (s2.role === 'heal') heal++; else { dd++; if (s2.type === 'magic') magic++; else phys++; }
      if (p.cls === lead.cls) same++;
      if (p.pet) pet++;
    });
    const OK = { tank: tank > 0, heal: heal > 0, dd1: dd > 0, dd2: dd > 1, phys2: phys > 1, magic2: magic > 1, same: same > 0, pet: pet > 0, two: comps.length > 1 };
    if (A) A.r.forEach(function (r) {
      if (!OK[r.k]) return;
      a.res.push(r.n);
      Object.keys(r.e || {}).forEach(function (f) {
        if (f === 'lowHp') { a.lowHp = (a.lowHp || 0) + 1; return; }
        a[f] = (a[f] || 0) + r.e[f];
      });
    });
    return a;
  }
