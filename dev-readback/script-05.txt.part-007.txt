, f = lvlFactor(level), base = {}, values = {};
    Object.keys(PRIMARY_NAMES).forEach(function (k) {
      base[k] = Math.max(1, Math.round((k === key ? 30 : 10) * f));
      const b = bonuses && Number(bonuses[k]);
      values[k] = Math.max(0, base[k] + (Number.isFinite(b) ? b : 0));
    });
    return { key: key, name: PRIMARY_NAMES[key], base: base, values: values,
      power: 0.5 + 0.5 * values[key] / base[key] };
  }
  function heroStats(cls, spec, level, bonuses) {
    const s = specOf(cls, spec), f = lvlFactor(level), primary = primaryStats(cls, spec, level, bonuses);
    const mana = resourceType(cls, spec) === 'mana' ? Math.round(70 + primary.values.intellect) : 0;
    const rawAtk = s.atk * f * ATKMUL * primary.power;
    const rawHeal = (s.heal || 0) * f * primary.power;
    return { mana: mana, hp: Math.round(s.hp * f + Math.max(0,Number(bonuses && bonuses.stamina)||0)*10), atk: Math.round(rawAtk), rawAtk: rawAtk,
      heal: Math.round(rawHeal), rawHeal: rawHeal, primary: primary, resource: resourceType(cls, spec),
      armor: 1-(1-s.armor)*(1-(Math.max(0,Number(bonuses && bonuses.armorFlat)||0)/(Math.max(0,Number(bonuses && bonuses.armorFlat)||0)+400+85*level))), spd: s.spd, role: s.role, name: s.name };
  }
