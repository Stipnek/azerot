  /* Урон и лечение героев по умениям. Считается из событий боя, сумма совпадает с итоговой статистикой. */
  function trackBd(evs) {
    let cur = null, skd = null;
    evs.forEach(function (e) {
      if (e.t === 'act') { cur = e.id; skd = null; }
      else if (e.t === 'auto') skd = { name: e.name, icon: autoEventIcon(e,st) };
      else if (e.t === 'skill') skd = B.SKILLS[e.skill];
      else if ((e.t === 'dmg' || e.t === 'heal') && e.amt > 0) {
        const src = B.byId(st, e.src); if (!src || src.side !== 'hero') return;
        const kind = e.t, per = e.dot || e.hot;
        const name = per ? (kind === 'dmg' ? 'Урон со временем' : 'Лечение со временем') : e.src === cur ? (skd ? skd.name : 'Обычная атака') : 'Прочее';
        const icon = per ? (kind === 'dmg' ? '☠️' : '🌿') : e.src === cur ? (skd ? wowIcon(skd) : '⚔️') : '✨';
        const h = bd[e.src] = bd[e.src] || { dmg: {}, heal: {} };
        const r = h[kind][name] = h[kind][name] || { v: 0, icon: icon };
        r.v += e.amt;
      }
    });
  }
  function bdHtml(hid, kind) {
    const m = (bd[hid] || {})[kind] || {}, rows = Object.keys(m).map(function (n) { return { n: n, v: m[n].v, i: m[n].icon }; }).sort(function (a, b) { return b.v - a.v; });
    const tot = rows.reduce(function (a, r) { return a + r.v; }, 0);
    if (!tot) return '<div class="brow"><span>Нет данных</span></div>';
    return rows.map(function (r) { return '<div class="brow"><span class="nm"><b>' + r.i + '</b><span>' + esc(r.n) + '</span></span><i>' + Math.round(r.v) + ' · ' + Math.round(r.v / tot * 100) + '%</i></div>'; }).join('');
  }
  function applyEvents(evs) {
    trackBd(evs);
    const actIdx = evs.findIndex(function (e) { return e.t === 'act'; });
    if (actIdx !== -1) {

      const sk = after.find(function (e) { return e.t === 'skill'; });
      const es = after.find(function (e) { return e.t === 'eskill'; });
      const au = after.find(function (e) { return e.t === 'auto'; }), adm = after.find(function (e) { return e.t === 'dmg' && e.auto; }), admSum = after.filter(function (e) { return e.t === 'dmg' && e.auto; }).reduce(function (a, e) { return a + e.amt + e.absorbed; }, 0);
      const nm = sk ? B.SKILLS[sk.skill].name : (es ? es.name : (au ? au.name : 'Атака'));
      const icon = sk ? wowIcon(B.SKILLS[sk.skill]) : (es && es.icon ? es.icon : (au ? autoEventIcon(au,st) : (u.side === 'hero' ? u.icon : '⚔️')));
      const dm = after.filter(function (e) { return e.t === 'dmg' && e.src === u.id && !e.dot && !e.auto; });

      let l2 = '', logTxt = '';

  }
