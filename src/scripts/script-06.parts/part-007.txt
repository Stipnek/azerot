    const t = toks[id]; if (!t || reduced) return;

  function hitFx(e) {
    if (!st) return;
    const tgt = B.byId(st, e.tgt); if (!tgt || !toks[e.tgt]) return;
    if (e.t === 'dmg') {
      const shown = e.miss ? 'Промах' : e.dodge ? 'Уклонение' : (e.amt ? ((tgt.side === 'hero' ? '−' : '') + Math.round(e.amt) + (e.absorbed ? ' (щит ' + Math.round(e.absorbed) + ')' : '')) : ('щит ' + Math.round(e.absorbed)));
      float(e.tgt, shown, (tgt.side === 'hero' && e.amt ? 'hurt' : '') + (e.crit ? ' crit' : '') + (e.dot ? ' tick' : ''));
      if (!e.dot && e.amt) setPose(e.tgt, 'hit', 340);
      if (!reduced && !e.dot && !e.dodge && !e.miss) toks[e.tgt].tc.animate([{ transform: 'translateX(0)', filter: 'brightness(1)' }, { transform: 'translateX(-5px)', filter: 'brightness(2) saturate(.3)' }, { transform: 'translateX(5px)' }, { transform: 'translateX(0)', filter: 'brightness(1)' }], { duration: 260 });
    } else if (e.t === 'heal' && e.amt > 0) {
      float(e.tgt, '+' + Math.round(e.amt), 'heal' + (e.crit ? ' crit' : '') + (e.hot ? ' tick' : ''));
      if (!e.hot) glow(e.tgt, '#6dff7a', 500);
    } else if ((e.t === 'buff' || e.t === 'debuff') && !reduced) {
      const col = e.t === 'buff' ? '#6aa6ff' : '#b46bff';
      glow(e.tgt, col, 550);
    }
  }

    const actIdx = evs.findIndex(function (e) { return e.t === 'act'; });

    if (actIdx === -1) return;
    const after = evs.slice(actIdx + 1);
    const actor = B.byId(st, evs[actIdx].id);
    if (!actor || !toks[actor.id]) return;
    if (actor.ghost) { after.forEach(function (e) { if (e.t === 'dmg' || e.t === 'heal') hitFx(e); }); return; }
    const sk = after.find(function (e) { return e.t === 'skill'; });
    const es = after.find(function (e) { return e.t === 'eskill'; });
    const icon = sk ? wowIcon(B.SKILLS[sk.skill]) : (es && es.icon ? es.icon : (actor.side === 'hero' ? '✦' : '✸'));
    const effects = after.filter(function (e) { return e.t === 'dmg' || (e.t === 'heal' && e.amt > 0) || e.t === 'buff' || e.t === 'debuff'; });
    const D = Math.min(760, 600 / speed * 0.72);
