  function foeLine(e) {

  /* Аналитика отряда: что даёт состав. Считается так же, как в бою: ауры по классам без повторов,
     ХП растёт от ауры ХП, урон от аур общего, физического и магического урона, получаемый урон снижается аурой защиты. */
  function openAnalytics() {
    if (!S.hero || !S.party.length) return;
    const keys = S.party.slice(), agg = B.leaderAura(keys.map(function (k) { return { cls: clsOf(k), spec: specKey(k), main: k === 'me', pet: petFor(k) }; }));
    const rows = keys.map(function (k) {
      const ck = clsOf(k), sp = B.specOf(ck, specKey(k)), hs = B.heroStats(ck, specKey(k), hLevel(k));
      const m = 1 + (agg.allDmg || 0) + (sp.type === 'phys' ? (agg.physDmg || 0) : sp.type === 'magic' ? (agg.magicDmg || 0) : 0);
      const hpE = Math.round(hs.hp * (1 + (agg.hp || 0)));
      return { k: k, name: heroName(k), ck: ck, sp: sp, role: sp.role, hpB: hs.hp, hpE: hpE, eff: Math.round(hpE / (1 - (agg.dmgTaken || 0))), atkB: hs.atk, atkE: Math.round(hs.atk * m), heal: Math.round((hs.heal || 0) * (1 + (agg.heal || 0))) };
    });
    const sum = function (f) { return rows.reduce(function (a, r) { return a + f(r); }, 0); };
    const hpB = sum(function (r) { return r.hpB; }), hpE = sum(function (r) { return r.eff; });
    const dB = sum(function (r) { return r.role === 'heal' ? 0 : r.atkB; }), dE = sum(function (r) { return r.role === 'heal' ? 0 : r.atkE; });
    const heal = sum(function (r) { return r.heal; });
    const pct = function (a, b) { return b ? Math.round((a / b - 1) * 100) : 0; };
    const up = function (a, b) { const p = pct(a, b); return p > 0 ? '<span class="up">+' + p + '%</span> <small>от ауры</small>' : '<span>ауры нет</span>'; };
    const mk = S.main + ':' + (B.CLASSES[S.main].specs[S.specs.me] ? S.specs.me : B.CLASSES[S.main].def), MA = B.AURAS[mk];
    const chips = [];
    if (agg.hp) chips.push('+' + Math.round(agg.hp * 100) + '% ХП');
    if (agg.allDmg) chips.push('+' + Math.round(agg.allDmg * 100) + '% урон всем');
    if (agg.physDmg) chips.push('+' + Math.round(agg.physDmg * 100) + '% физ. урон');
    if (agg.magicDmg) chips.push('+' + Math.round(agg.magicDmg * 100) + '% магический урон');
    if (agg.dmgTaken) chips.push('−' + Math.round(agg.dmgTaken * 100) + '% получаемый урон');
    if (agg.crit) chips.push('+' + Math.round(agg.crit * 100) + '% шанс крита');
    if (agg.heal) chips.push('+' + Math.round(agg.heal * 100) + '% сила лечения');
    if (agg.spd) chips.push('+' + Math.round(agg.spd * 100) + '% скорость');
    if (agg.lowHp) chips.push('урон растёт при потере ХП');
    const auraRows = agg.name ? '<div class="arow"><div class="an-t">✦ ' + esc(agg.name) + ' <small>' + esc(B.CLASSES[S.main].name) + ', спек вашего героя</small></div>' + chips.map(function (c) { return '<span class="acp g">' + esc(c) + '</span>'; }).join('') +
      (MA ? '<br><small>' + MA.r.map(function (r) { return (agg.res.indexOf(r.n) !== -1 ? '✓ <b>' : '— <b>') + esc(r.n) + '</b> (' + esc(r.w) + ': ' + esc(r.x) + ')'; }).join(' · ') + '</small>' : '') + '</div>' : '';
    const maxHp = Math.max.apply(null, rows.map(function (r) { return r.eff; })), maxAtk = Math.max.apply(null, rows.map(function (r) { return Math.max(r.atkE, r.heal); }));
    const heroRows = rows.map(function (r) {
      const gainHp = r.eff - r.hpB, main = r.role === 'heal' ? r.heal : r.atkE, mainB = r.role === 'heal' ? r.heal : r.atkB;
      return '<div class="hrow"><div class="hn"><span><b>' + esc(r.name) + '</b> ' + esc(B.CLASSES[r.ck].name) + ', ' + RN[r.role] + '</span><span>' + (S.party.indexOf(r.k) === 0 ? 'вы' : '') + '</span></div>' +
        '<div class="bar" title="ХП"><i class="b" style="width: ' + Math.round(r.hpB / maxHp * 100) + '%"></i><i class="g" style="width: ' + Math.max(0, Math.round(gainHp / maxHp * 100)) + '%"></i></div><div class="bl"><span>Живучесть ' + r.eff + (gainHp > 0 ? ' (+' + gainHp + ')' : '') + '</span><span>ХП ' + r.hpB + '</span></div>' +
        '<div class="bar" title="' + (r.role === 'heal' ? 'Лечение' : 'Урон') + '"><i class="b" style="width: ' + Math.round(mainB / maxAtk * 100) + '%"></i><i class="g" style="width: ' + Math.max(0, Math.round((main - mainB) / maxAtk * 100)) + '%"></i></div><div class="bl"><span>' + (r.role === 'heal' ? 'Лечение ' + r.heal : 'Урон ' + r.atkE + (r.atkE > r.atkB ? ' (+' + (r.atkE - r.atkB) + ')' : '')) + '</span><span>' + (r.role === 'heal' ? '' : 'база ' + r.atkB) + '</span></div></div>';
    }).join('');
    const roles = { tank: 0, heal: 0, dd: 0 }; rows.forEach(function (r) { roles[r.role]++; });
    const warns = [];
    if (!roles.tank) warns.push('Нет танка: враги бьют тех, кто наносит больше всего урона.');
    if (!roles.heal) warns.push('Нет лекаря: урон никто не восполнит.');
    $('pop-c').innerHTML = '<div class="an"><p class="sect">Аналитика отряда</p><p class="hint">Что даёт состав: ' + rows.length + ' героев. Числа как в бою, с учётом уровня, ауры лидера и резонансов. Процент: на сколько больше, чем у этого же отряда без ауры лидера.</p>' +
      '<div class="mets"><div class="met"><small>Живучесть</small><b>' + hpE + '</b>' + up(hpE, hpB) + '</div><div class="met"><small>Урон за удар</small><b>' + dE + '</b>' + up(dE, dB) + '</div><div class="met"><small>Лечение</small><b>' + heal + '</b><span>' + 'танки ' + roles.tank + ', лекари ' + roles.heal + ', бойцы ' + roles.dd + '</span></div></div>' +
      '<h4>Аура лидера</h4>' + auraRows + '<h4>Герои</h4>' + heroRows + warns.map(function (w) { return '<p class="warn">' + esc(w) + '</p>'; }).join('') + '</div>';
    $('pop').hidden = false;
  }
  /* Удержание 0,8 секунды вверху экрана «Отряд» (палец или мышь) открывает аналитику; вокруг точки касания рисуется кольцо по часовой стрелке. */
  (function () {
    let tm = null, sx = 0, sy = 0, fired = false;
    const hb = function () { return $('holdring'); };
    const stop = function () { clearTimeout(tm); tm = null; const b = hb(); if (b) { b.classList.remove('run'); b.hidden = true; } };
    $('prep').addEventListener('pointerdown', function (ev) {
      if (!ev.target.closest('#pt-head') || ev.target.closest('button')) return;
      sx = ev.clientX; sy = ev.clientY; fired = false;
      const b = hb(); b.style.left = ev.clientX + 'px'; b.style.top = ev.clientY + 'px'; b.classList.remove('run'); b.hidden = false; void b.offsetWidth; b.classList.add('run');
      clearTimeout(tm);
      tm = setTimeout(function () { fired = true; stop(); if (navigator.vibrate) navigator.vibrate(30); openAnalytics(); }, 800);
    });
    ['pointerup', 'pointercancel', 'pointerleave'].forEach(function (n) { $('prep').addEventListener(n, stop); });
    $('prep').addEventListener('pointermove', function (ev) {
      if (!tm) return;
      if (Math.hypot(ev.clientX - sx, ev.clientY - sy) > 40) { stop(); return; }
      const b = hb(); b.style.left = ev.clientX + 'px'; b.style.top = ev.clientY + 'px';
    });
    $('prep').addEventListener('contextmenu', function (ev) { if (ev.target.closest('#pt-head')) ev.preventDefault(); });
    $('prep').addEventListener('click', function (ev) { if (fired) { fired = false; ev.stopPropagation(); ev.preventDefault(); } }, true);
  })();
