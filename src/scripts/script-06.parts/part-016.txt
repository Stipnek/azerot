  function startCrGallery(race, cls, hostId) {
    const host = $(hostId || 'cr-stage'), epoch = crGalleryEpoch, keys = cls ? AZ_ASSET_REGISTRY.raceClassGallery[race+':'+cls] || [] : AZ_ASSET_REGISTRY.creationGalleryByRace[race] || [];
    window.AzerotArtDiagnostics = window.AzerotArtDiagnostics || {};
    const diagnostics = window.AzerotArtDiagnostics[(hostId || 'cr-stage') + ':' + (cls || race)] = {events:[]};
    const layers = Array.from(host.querySelectorAll('.cr46-art')), status = host.querySelector('.cr46-art-status');
    const pause = host.querySelector('[data-cr46-pause]'), next = host.querySelector('[data-cr46-next]');
    const source = host.querySelector('.cr46-source'), counter = host.querySelector('.cr46-art-count');
    const reduced = window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches;
    let current = -1, active = 0, busy = false, failed = new Set();
    function valid() { return epoch === crGalleryEpoch && host.isConnected && !host.closest('[hidden]'); }
    if(!keys.length){ status.textContent='Подходящая иллюстрация ещё не добавлена';pause.hidden=true;next.hidden=true;counter.textContent='';return; }
    function schedule() {
      clearTimeout(crGalleryTimer);
      if (valid() && !crGalleryPaused && !reduced && keys.length > 1 && failed.size < keys.length - 1)
        crGalleryTimer = setTimeout(function () { show((current + 1) % keys.length, 0); }, 10000);
    }
    function show(index, attempts) {
      if (!valid() || busy || !keys.length) return;
      if (attempts >= keys.length) { status.textContent = 'Иллюстрации недоступны. Выбор расы работает.'; host.classList.add('cr46-art-error'); return; }
      const asset = AZ_ASSET_REGISTRY.assets[keys[index]];
      if (!asset || asset.race !== race || (cls && asset.characterClass !== cls)) { diagnostics.events.push({status:'blocked_identity_mismatch',key:keys[index]}); failed.add(index); show((index + 1) % keys.length, attempts + 1); return; }
      busy = true;
      const preload = new Image(); preload.referrerPolicy = 'no-referrer';
      const urls = [asset.url].concat(asset.fallbackUrls || []); let candidate = 0;
      let settled = false;
      let timeout = setTimeout(failure, 12000);
      function failure() {
        if (settled) return; settled = true; clearTimeout(timeout); busy = false;
        if (!valid()) return;
        if (candidate + 1 < urls.length) {
          diagnostics.events.push({status:'candidate_failed',url:urls[candidate]});
          settled = false; busy = true; timeout = setTimeout(failure,12000); preload.src = urls[++candidate]; return;
        }
        diagnostics.events.push({status:'load_error',key:keys[index]});
        failed.add(index);
        if (current >= 0) { status.textContent = 'Часть иллюстраций недоступна'; schedule(); }
        else show((index + 1) % keys.length, attempts + 1);
      }
      preload.onerror = failure;
      preload.onload = function () {
        const decoded = preload.decode ? preload.decode() : Promise.resolve();
        decoded.then(function () {
          if (settled) return; settled = true; clearTimeout(timeout); busy = false;
          if (!valid()) return;
          failed.delete(index);
          const target = current < 0 ? active : 1 - active;
          layers[target].src = urls[candidate]; layers[target].alt = asset.name + (cls ? ' — иллюстрация класса' : ' — иллюстрация расы');
          diagnostics.current = {key:keys[index],url:urls[candidate],width:preload.naturalWidth,height:preload.naturalHeight};
          layers[target].classList.add('visible');
          if (current >= 0) layers[active].classList.remove('visible');
          active = target; current = index;
          host.classList.add('cr46-has-art'); host.classList.remove('cr46-art-error');
          status.textContent = asset.name + ' · Hearthstone / Blizzard';
          counter.textContent = (index + 1) + ' / ' + keys.length;
          source.href = asset.source; source.hidden = false;
          schedule();
        }).catch(failure);
      };
      preload.src = urls[candidate];
    }
    function pauseLabel() {
      pause.textContent = crGalleryPaused || reduced ? 'Продолжить' : 'Пауза';
      pause.setAttribute('aria-pressed', String(crGalleryPaused || reduced));
      next.disabled = keys.length < 2;
      if (keys.length < 2) {pause.textContent = 'Иллюстрация';pause.disabled = true;}
      if (reduced) { pause.textContent = 'Без анимации'; pause.disabled = true; }
    }
    pause.addEventListener('click', function () { crGalleryPaused = !crGalleryPaused; pauseLabel(); if (crGalleryPaused) clearTimeout(crGalleryTimer); else schedule(); });
    next.addEventListener('click', function () { clearTimeout(crGalleryTimer); show((current + 1) % keys.length, 0); });
    pauseLabel(); show(0, 0);
  }
