  function walkTo(pl, done, stopFrac) {
    const c = S.hero, pts = routeTo(pl), seg = [];
    let total = 0;
    for (let i = 1; i < pts.length; i++) { const d = pdist(pts[i - 1], pts[i]); seg.push(d); total += d; }
    const end = function (p, at) { c.pos = { x: p.x, y: p.y }; c.at = at; S.walk = null; save(); };
    if (total < 1) { end(pts[pts.length - 1], pl); done(false); return; }
    const goal = stopFrac ? total * stopFrac : total;
    const ms = Math.max(900, Math.min(2200, goal * 6));
    const cut = [pts[0]];
    let acc = 0;
    for (let i = 0; i < seg.length; i++) {
      if (acc + seg[i] >= goal - 0.01) { const t = seg[i] ? Math.min(1, (goal - acc) / seg[i]) : 1; cut.push({ x: pts[i].x + (pts[i + 1].x - pts[i].x) * t, y: pts[i].y + (pts[i + 1].y - pts[i].y) * t }); break; }
      acc += seg[i]; cut.push(pts[i + 1]);
    }
    const last = cut[cut.length - 1];
    S.qopen = false; S.hub = null; renderMap();
    const map = $('map'), tp = PLACES[pl];
    map.classList.add('walking');
    map.insertAdjacentHTML('beforeend', '<svg class="route" viewBox="0 0 100 100" preserveAspectRatio="none" aria-hidden="true"><polyline points="' + pts.map(function (p) { return p.x + ',' + p.y; }).join(' ') + '"/></svg><span class="dest" style="left: ' + tp.x + '%; top: ' + tp.y + '%"></span><div class="wcap" role="status">Идём: ' + esc(tp.label || tp.name) + ' · тап — пропустить</div>');
    const me = $('me'); if (me) me.classList.add('mw');
    window.scrollTo(0, 0);
    let run2 = 0, an = null;
    const kf = cut.map(function (p, i) { if (i) run2 += pdist(cut[i - 1], p); return { left: p.x + '%', top: p.y + '%', offset: Math.min(1, run2 / goal) }; });
    kf[kf.length - 1].offset = 1;
    const fin = function () {
      if (!S.walk) return;
      if (an) { an.onfinish = null; an.cancel(); }
      if (me) { me.style.left = last.x + '%'; me.style.top = last.y + '%'; me.classList.remove('mw'); }
      map.classList.remove('walking');
      end(last, stopFrac ? null : pl); done(!!stopFrac);
    };
    S.walk = { skip: fin };
    if (me && me.animate) { an = me.animate(kf, { duration: ms, easing: 'linear', fill: 'forwards' }); an.onfinish = fin; }
    else setTimeout(fin, 300);
  }
