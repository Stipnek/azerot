  function polyMetrics() {
    return st.heroes.filter(function (h) { return !h.dummy; }).map(function (h) {
      const turns = Math.max(1, h.stats.turns || 0), per = new Array(turns).fill(0), mine = st.series.filter(function (x) { return x[1] === h.id; });
      let crit = 0; mine.forEach(function (x) { const i = Math.min(turns, Math.max(1, x[0])) - 1; per[i] += x[2]; crit += x[3]; });
      const total = per.reduce(function (a, b) { return a + b; }, 0), W = Math.min(5, turns);
      let peak = 0; for (let i = 0; i + W <= turns; i++) { let sm = 0; for (let j = 0; j < W; j++) sm += per[i + j]; peak = Math.max(peak, sm / W); }
      const f5 = per.slice(0, W).reduce(function (a, b) { return a + b; }, 0) / W;
      const W10 = Math.min(10, turns), f10 = per.slice(0, W10).reduce(function (a, b) { return a + b; }, 0) / W10;
      const hl = st.hlog.filter(function (x) { return x[1] === h.id; }), heal = hl.reduce(function (a, x) { return a + x[2]; }, 0), over = hl.reduce(function (a, x) { return a + x[3]; }, 0);
      return { id: h.id, key: h.cls + '.' + h.spec, cls: h.cls, name: h.name, specName: h.specName || '', icon: h.icon, role: h.role, turns: turns, total: total, avg: total / turns, peak: peak, first5: f5, first10: f10, crit: mine.length ? crit / mine.length : 0, heal: heal, healAvg: heal / turns, overPct: heal + over ? over / (heal + over) : 0, taken: h.stats.taken, dead: h.hp <= 0, per: per };
    });
  }
  function polyChart(rows) {
    const sm = rows.map(function (r) { return r.per.map(function (v, i, a) { const w = a.slice(Math.max(0, i - 2), i + 1); return w.reduce(function (x, y) { return x + y; }, 0) / w.length; }); });
    const N = Math.max.apply(null, rows.map(function (r) { return r.per.length; })), mx = Math.max(1, Math.max.apply(null, sm.map(function (a) { return Math.max.apply(null, a.concat([0])); })));
    if (N < 2) return '';
    const lines = rows.map(function (r, k) { return '<polyline fill="none" stroke-width="2" stroke="var(--c-' + r.cls + ')" points="' + sm[k].map(function (v, i) { return (8 + i / (N - 1) * 284).toFixed(1) + ',' + (96 - v / mx * 84).toFixed(1); }).join(' ') + '"/>'; }).join('');
    return '<svg class="pchart" viewBox="0 0 300 104" role="img" aria-label="График урона по ходам"><line x1="8" y1="96" x2="292" y2="96" stroke="#8a7a5a" stroke-width="1"/><text x="8" y="10" font-size="9" fill="#bfae86">урон за ход, пик ' + Math.round(mx) + '</text>' + lines + '</svg>';
  }
