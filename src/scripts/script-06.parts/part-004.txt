    const keys = S.hero ? S.party.slice() : battleParty().slice();

    const party = keys.map(function (key) { const g = tavRec(key); return {

      race: key === 'me' ? S.hero && S.hero.race : g && g.race,
      statBonuses:equipmentStats(key), rb: tavBonus(g), pet: petFor(key)
    }; });
    return B.createBattle({ party: party, main: S.main, encounter: 'test', level: S.level, seed: 1 }).heroes[keys.indexOf(k)];
