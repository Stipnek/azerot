  function levelColor(partyLvl, lvl) {
    const d = lvl - partyLvl;
    if (d >= 5) return 'red';
    if (d >= 3) return 'orange';
    if (d >= -2) return 'yellow';
    if (d >= -5) return 'green';
    return 'grey';
  }
  /* Опыт: на 1 уровень 350, дальше +10% от 350 за уровень; награда 100, +5% от 100 за уровень героя */
  const XPC = { first: 350, step: 0.10, base: 100, rew: 0.05, cap: 30, talk: 0.6, elite: 2.5, group: 2, room: 1, mainPct: 1, fightPct: 1.25, rosterPct: 1 };
  function xpNeed(l) { return Math.round(XPC.first * (1 + XPC.step * (l - 1))); }
  function xpAt(l) { let s = 0; for (let i = 1; i < l; i++) s += xpNeed(i); return s; }
  function levelOf(xp) { let l = 1, x = Math.max(0, xp); while (l < XPC.cap && x >= xpNeed(l)) { x -= xpNeed(l); l++; } return l; }
  function levelProgress(xp) { const l = levelOf(xp); return l >= XPC.cap ? { level: l, cur: 0, need: 0 } : { level: l, cur: Math.max(0, xp) - xpAt(l), need: xpNeed(l) }; }
  function xpFor(heroLvl, enc) {
    if (enc.rooms) return enc.rooms.reduce(function (a, r) { return a + xpFor(heroLvl, ENCOUNTERS[r]); }, 0);
    const d = enc.level - heroLvl;
    const mult = enc.xpMult != null ? enc.xpMult : (enc.type === 'talk' ? XPC.talk : enc.type === 'dungeon' ? XPC.room : enc.elite ? XPC.elite : 1);
    const m = d <= -6 ? 0 : d <= -3 ? 0.5 : d > 0 ? 1 + 0.1 * Math.min(d, 6) : 1;
    const grp = enc.size >= 3 && enc.type !== 'talk' ? XPC.group : 1; /* групповые бои: ×2 опыта (решение 16 ноября) */
    return Math.round(XPC.base * (1 + XPC.rew * (heroLvl - 1)) * mult * grp * m);
  }
  /* кто сколько получил за квест: мейн 100%, компаньон в бою 125% (пока ниже мейна), остальной ростер 25% */
  function xpShares(enc, heroes) {
    const mainLv = (heroes.find(function (h) { return h.main; }) || { level: 1 }).level;
    return heroes.map(function (h) {
      const k = h.main ? XPC.mainPct : (h.fights ? (h.level < mainLv ? XPC.fightPct : 1) : XPC.rosterPct);
      return { cls: h.cls, xp: Math.round(xpFor(h.level, enc) * k) };
    });
  }

  const PRIMARY_NAMES = { strength: 'Сила', agility: 'Ловкость', intellect: 'Интеллект' };
  function primaryKey(cls, spec) {
    if (cls === 'warrior' || (cls === 'paladin' && spec !== 'holy')) return 'strength';
    if (cls === 'hunter' || cls === 'rogue' || (cls === 'druid' && (spec === 'bear' || spec === 'cat'))) return 'agility';
    return 'intellect';
  }
  function resourceType(cls, spec) {
    if (cls === 'warrior' || (cls === 'druid' && spec === 'bear')) return 'rage';
    if (cls === 'rogue' || (cls === 'druid' && spec === 'cat')) return 'energy';
    return 'mana';
  }
  function primaryStats(cls, spec, level, bonuses) {
    const key = primaryKey(cls, spec), f = lvlFactor(level), base = {}, values = {};
    Object.keys(PRIMARY_NAMES).forEach(function (k) {
      base[k] = Math.max(1, Math.round((k === key ? 30 : 10) * f));
      const b = bonuses && Number(bonuses[k]);
      values[k] = Math.max(0, base[k] + (Number.isFinite(b) ? b : 0));
    });
    return { key: key, name: PRIMARY_NAMES[key], base: base, values: values,
      power: 0.5 + 0.5 * values[key] / base[key] };
  }
  function heroStats(cls, spec, level, bonuses) {
    const s = specOf(cls, spec), f = lvlFactor(level), primary = primaryStats(cls, spec, level, bonuses);
    const mana = resourceType(cls, spec) === 'mana' ? Math.round(70 + primary.values.intellect) : 0;
    const rawAtk = s.atk * f * ATKMUL * primary.power;
    const rawHeal = (s.heal || 0) * f * primary.power;
    return { mana: mana, hp: Math.round(s.hp * f + Math.max(0,Number(bonuses && bonuses.stamina)||0)*10), atk: Math.round(rawAtk), rawAtk: rawAtk,
      heal: Math.round(rawHeal), rawHeal: rawHeal, primary: primary, resource: resourceType(cls, spec),
      armor: 1-(1-s.armor)*(1-(Math.max(0,Number(bonuses && bonuses.armorFlat)||0)/(Math.max(0,Number(bonuses && bonuses.armorFlat)||0)+400+85*level))), spd: s.spd, role: s.role, name: s.name };
  }

  function leaderAura(party) {
    const a = { res: [] };
    const ps = (party || []).filter(function (p) { return p && CLASSES[p.cls]; });
    const lead = ps.find(function (p) { return p.main; }) || ps[0];
    if (!lead) return a;
    const c = CLASSES[lead.cls], specKey = c.specs[lead.spec] ? lead.spec : c.def;
    const A = AURAS[lead.cls + ':' + specKey];
    if (A) {
      ['physDmg', 'magicDmg', 'allDmg', 'hp', 'dmgTaken', 'crit', 'heal', 'spd'].forEach(function (f) { if (A.b[f]) a[f] = A.b[f]; });
      if (A.b.lowHp) a.lowHp = 1;
      a.name = A.n; a.text = A.t;
    }
    const comps = ps.filter(function (p) { return p !== lead; });
    let tank = 0, heal = 0, dd = 0, phys = 0, magic = 0, same = 0, pet = 0;
    comps.forEach(function (p) {
      const c2 = CLASSES[p.cls], s2 = c2.specs[c2.specs[p.spec] ? p.spec : c2.def];
      if (s2.role === 'tank') tank++; else if (s2.role === 'heal') heal++; else { dd++; if (s2.type === 'magic') magic++; else phys++; }
      if (p.cls === lead.cls) same++;
      if (p.pet) pet++;
    });
    const OK = { tank: tank > 0, heal: heal > 0, dd1: dd > 0, dd2: dd > 1, phys2: phys > 1, magic2: magic > 1, same: same > 0, pet: pet > 0, two: comps.length > 1 };
    if (A) A.r.forEach(function (r) {
      if (!OK[r.k]) return;
      a.res.push(r.n);
      Object.keys(r.e || {}).forEach(function (f) {
        if (f === 'lowHp') { a.lowHp = (a.lowHp || 0) + 1; return; }
        a[f] = (a[f] || 0) + r.e[f];
      });
    });
    return a;
  }

  function racialAct(st, h) {
    const r = h.racial; if (!r || h.cd[r]) return;
    const foes = alive(st.enemies); if (!foes.length) return;
    const fromFoe = function (f) { const o = byId(st, f.src); return o && o.side === 'enemy'; };
    const go = function () { h.cd[r] = SKILLS[r].cd; emit(st, { t: 'skill', src: h.id, skill: r }); };
    if (r === 'r_emfh') {
      // WotLK exception: remove existing movement/control effects, never grant immunity.
      const isRacialControl = function (f) { return ['stun', 'root', 'slow', 'silence'].indexOf(f.kind) !== -1; };
      if (h.effects.some(isRacialControl)) { go(); h.effects = h.effects.filter(function (f) { return !isRacialControl(f); }); emit(st, { t: 'buff', tgt: h.id }); }
    } else if (r === 'r_stoneform') {
      if (h.effects.some(function (f) { return f.kind === 'dot' && fromFoe(f); }) || pct(h) < 0.55) { go(); h.effects = h.effects.filter(function (f) { return !(f.kind === 'dot' && fromFoe(f)); }); addEffect(h, { kind: 'armorup', amt: 0.10, turns: 3, src: h.id, name: 'Каменная форма', icon: '🗿' }); emit(st, { t: 'buff', tgt: h.id }); }
    } else if (r === 'r_escape') {
      if (h.effects.some(function (f) { return (f.kind === 'slow' || f.kind === 'root') && fromFoe(f); })) { go(); h.effects = h.effects.filter(function (f) { return !((f.kind === 'slow' || f.kind === 'root') && fromFoe(f)); }); emit(st, { t: 'buff', tgt: h.id }); }
    } else if (r === 'r_meld') {
      if (pct(h) < 0.7) { go(); addEffect(h, { kind: 'dodge', amt: 0.35, turns: 3, src: h.id, name: 'Слияние с тенью', icon: '👤' }); emit(st, { t: 'buff', tgt: h.id }); }
    }
  }

  function targetsOf(st) { let hs = alive(st.heroes); const vis = hs.filter(function (h) { return !h.effects.some(function (f) { return f.kind === 'hidden'; }); }); if (vis.length) hs = vis; return hs.concat(st.petTargets ? alive(st.pets || []) : []); }
  function createPet(st, owner, key, spec, f, aura) {
    const d = PETS[key]; if (!d) return null;

      maxHp, hp: maxHp, atk: spec.atk * d.atk * f * ATKMUL, armor: d.armor, res: d.res, spd: d.spd * (1 + (aura.spd || 0)), crit: d.crit + (aura.crit || 0), threatMult: 1, type: d.type, healPow: 0,
      skills: d.skills.slice(), cd: {}, effects: [], tm: st.rnd() * 150, isMain: false, maxMana: 0, mana: 0, oom: false, oomHit: false, maxCp: 0, cp: 0, stats: { dmg: 0, heal: 0, taken: 0 }, dead: false, reviveIn: 0 };

  function petsTick(st, owner) {
    (st.pets || []).forEach(function (p) {
      if (p.owner !== owner.id || p.gone) return;
      if (p.hp > 0) { p.dead = false; return; }
      if (!p.dead) { p.dead = true; p.reviveIn = PET_REVIVE.turns; log(st, p.name + ' погиб и вернётся через ' + PET_REVIVE.turns + ' хода', 'info'); return; }
      p.reviveIn--;
      if (p.reviveIn <= 0) {
        p.hp = Math.round(p.maxHp * PET_REVIVE.hp); p.dead = false; p.effects = []; p.cd = {};
        emit(st, { t: 'revive', id: p.id }); log(st, p.name + ' возвращается в бой', 'good');
      }
    });
  }
  function petAct(st, p) {
    const foes = alive(st.enemies); if (!foes.length) return;
    const owner = byId(st, p.owner), t = focusEnemy(st);
    const sk = function (s) { return p.skills.indexOf(s) !== -1 && !p.cd[s]; };
    const usep = function (s) { p.cd[s] = SKILLS[s].cd; emit(st, { t: 'skill', src: p.id, skill: s }); };
    const weakest = foes.filter(function (e) { return !e.boss && !e.effects.some(function (f) { return f.kind === 'stun' || f.kind === 'silence'; }); }).sort(function (a, b) { return b.atk - a.atk; })[0];
    const d = PETS[p.spec];
    emit(st, { t: 'auto', src: p.id, name: d.auto.name, icon: d.auto.icon });
    damage(st, p, t, PET_AUTO, { type: d.type, auto: true, threatBonus: p.spec === 'voidwalker' ? 2 : 1 });
    if (!alive(st.enemies).length) return;
    const tt = focusEnemy(st);
    switch (p.spec) {
      case 'imp':
        if (sk('pfireshield') && owner && owner.hp > 0 && pct(owner) < 0.85 && !owner.effects.some(function (f) { return f.kind === 'reflect'; })) { usep('pfireshield'); addEffect(owner, { kind: 'reflect', amt: 0.3, turns: 3, src: p.id, name: 'Огненный щит', icon: '🔥' }); emit(st, { t: 'buff', tgt: owner.id }); return; }
        if (sk('pfireball')) { usep('pfireball'); damage(st, p, tt, 1.3, { type: 'magic' }); }
        return;
      case 'voidwalker':
        if (sk('pconsume') && pct(p) < 0.5) { usep('pconsume'); heal(st, p, p, p.maxHp * 0.3, { noCrit: true }); return; }
        if (sk('ptorment')) { usep('ptorment'); damage(st, p, tt, 1.0, { threatBonus: 3 }); }
        return;
      case 'felhunter':
        if (sk('pspelllock') && weakest) { usep('pspelllock'); addEffect(weakest, { kind: 'silence', turns: 2, src: p.id, name: 'Блокировка заклинаний', icon: '🚫' }); emit(st, { t: 'debuff', tgt: weakest.id }); return; }
        if (sk('pfelbite')) { usep('pfelbite'); damage(st, p, tt, 1.3); }
        return;
      case 'succubus':
        if (sk('pseduction') && weakest) { usep('pseduction'); addEffect(weakest, { kind: 'stun', turns: 2, src: p.id, name: 'Соблазнение', icon: '💋' }); emit(st, { t: 'debuff', tgt: weakest.id }); return; }
        if (sk('plash')) { usep('plash'); damage(st, p, tt, 1.4, { type: 'magic' }); }
        return;
      case 'wolf':
        if (sk('phowl') && owner && owner.hp > 0 && !owner.effects.some(function (f) { return f.kind === 'howl'; })) { usep('phowl'); alive(st.heroes).forEach(function (h) { addEffect(h, { kind: 'howl', turns: 3, src: p.id, name: 'Яростный вой', icon: '📢' }); }); return; }
        if (sk('pbite')) { usep('pbite'); damage(st, p, tt, 1.3); }
        return;
    }
  }

  function alive(list) { return list.filter(function (u) { return u.hp > 0; }); }

  function refreshAggro(st, e) {
    const taunt = e.effects.find(function (f) { return f.kind === 'taunt'; });
    if (taunt) { const t = byId(st, taunt.target); if (t && t.hp > 0) { e.aggro = t.id; return; } }
    let best = null, bestV = 0;
    targetsOf(st).forEach(function (h) { const v = e.threat[h.id] || 0; if (v > bestV) { bestV = v; best = h; } });
    e.aggro = best ? best.id : null;
  }

  function lowHpBonus(p) { if (p > 0.9) return 0; return Math.min(9, Math.floor((0.9 - p) / 0.1) + 1) * 0.05; }
  function armorMod(u) { const f = u.effects.find(function (x) { return x.kind === 'armorbreak'; }); return f ? 1 - 0.14 * f.stacks : 1; }
  function breakArmor(src, tgt) {
    const f = tgt.effects.find(function (x) { return x.kind === 'armorbreak'; });
    if (f) { f.stacks = Math.min(5, f.stacks + 1); f.turns = 4; f.name = 'Снятая броня ×' + f.stacks; }
    else tgt.effects.push({ kind: 'armorbreak', stacks: 1, turns: 4, src: src.id, name: 'Снятая броня ×1', icon: '🔻' });
  }
  function dmgMult(st, src) {
    if (src.side !== 'hero') {
      let pm = 1;
      src.effects.forEach(function (f) { if (f.kind === 'power') pm *= f.mult; });
      return (hasEff(src, 'enrage') ? 1.5 : 1) * pm * (1 + (st.enrage ? st.enrageStacks * st.enrage.step : 0));
    }
    const a = st.aura;
    let m = 1 + (a.allDmg || 0);
    if (a.lowHp) m += lowHpBonus(src.hp / src.maxHp) * a.lowHp;
    if (src.type === 'phys') m += a.physDmg || 0;
    if (src.type === 'magic') m += a.magicDmg || 0;
    return m;
  }

  // v30: faster damage-based rage; bear gains extra rage only when taking damage.
  // v31: provisional turn-based energy; costs are deliberately not active yet.
  const ENERGY_CFG = Object.freeze({ max: 100, start: 100, regen: 35 });
  const RAGE_CFG = Object.freeze({ max: 100, start: 0, autoFactor: 15, takenFactor: 5, bearAutoFactor: 7.5, takenPerTurn: 25 });
  function rageConversion(level) {
    const l = Math.max(1, Math.min(60, Number.isFinite(level) ? level : 1));
    return 0.0091107836 * l * l + 3.225598133 * l + 4.2652911;
  }
  function gainRage(u, amount, source) {
    if (!u || u.resource !== 'rage' || !u.maxRage || u.hp <= 0 || !Number.isFinite(amount) || amount <= 0) return 0;
    const before = Number.isFinite(u.rage) ? Math.max(0, Math.min(u.maxRage, u.rage)) : 0;
    let grant = amount;
    if (source === 'taken') grant = Math.min(grant, Math.max(0, RAGE_CFG.takenPerTurn - (u.rageTakenThisTurn || 0)));
    grant = Math.max(0, Math.min(grant, u.maxRage - before));
    u.rage = before + grant;
    if (source === 'taken') u.rageTakenThisTurn = (u.rageTakenThisTurn || 0) + grant;
    return grant;
  }

  function damage(st, src, tgt, mult, opt) {
    opt = opt || {};
    if (tgt.hp <= 0) return 0;
    const type = opt.type || src.type || 'phys';
    let blocked = false;
    if (tgt.ghost) return 0;
    if (!st.test && !opt.refl && src.side !== tgt.side) {
      const ld = src.side === 'hero' ? st.encLevel - src.level : tgt.level - st.encLevel;
      if (ld > 0 && st.rnd() < lvlMiss(ld)) {
        emit(st, { t: 'dmg', src: src.id, tgt: tgt.id, amt: 0, absorbed: 0, crit: false, dot: !!opt.dot, auto: !!opt.auto, miss: true });
        return 0;
      }
    }
    if (tgt.side === 'hero' && src.side === 'enemy' && !opt.dot && !opt.refl && type !== 'magic') {
      const ms = src.effects.reduce(function (a, f) { return a + (f.kind === 'miss' ? f.amt : 0); }, 0);
      if (ms > 0 && st.rnd() < ms) { emit(st, { t: 'dmg', src: src.id, tgt: tgt.id, amt: 0, absorbed: 0, crit: false, dot: false, auto: !!opt.auto, miss: true }); log(st, src.name + ' промахивается', 'good'); return 0; }
      const dg = tgt.effects.reduce(function (a, f) { return a + (f.kind === 'dodge' ? f.amt : 0); }, 0) + (tgt.dodgeBase || 0);
      if (dg > 0 && st.rnd() < dg) { emit(st, { t: 'dmg', src: src.id, tgt: tgt.id, amt: 0, absorbed: 0, crit: false, dot: false, auto: !!opt.auto, dodge: true }); log(st, tgt.name + ' уклоняется от удара', 'good'); if (tgt.skills && tgt.skills.indexOf('revenge') !== -1) tgt.revengeReady = PROT_CFG.revengeTurns; return 0; }
      if (tgt.parryBase > 0 && st.rnd() < tgt.parryBase) {
        if (tgt.skills && tgt.skills.indexOf('revenge') !== -1) tgt.revengeReady = PROT_CFG.revengeTurns;
        emit(st, {t:'dmg',src:src.id,tgt:tgt.id,amt:0,absorbed:0,crit:false,dot:false,auto:!!opt.auto,parry:true}); log(st,tgt.name+' парирует удар','good'); return 0;
      }
      if (tgt.ai === 'warrior_prot' && st.rnd() < PROT_CFG.blockChance) {
        blocked = true; tgt.revengeReady = PROT_CFG.revengeTurns;
        emit(st,{t:'block',src:src.id,tgt:tgt.id}); log(st,tgt.name+' блокирует удар щитом','good');
      }
    }
    let amt = src.atk * mult * dmgMult(st, src) * (0.9 + st.rnd() * 0.2);
    const cbEff = (!opt.dot && !opt.refl && !opt.auto && src.effects) ? src.effects.find(function (f) { return f.kind === 'coldblood'; }) : null;
    if (cbEff) src.effects = src.effects.filter(function (f) { return f !== cbEff; });
    if (src.side === 'hero' && !opt.dot && !opt.refl && src.effects && src.effects.some(function (f) { return f.kind === 'hidden'; })) src.effects = src.effects.filter(function (f) { return f.kind !== 'hidden'; });
    const crit = !!cbEff || !opt.noCrit && st.rnd() < (src.crit || 0) + (src.effects && src.effects.some(function (f) { return f.kind === 'combust'; }) ? 0.4 : 0) + (opt.critBonus || 0) + (src.shatter && tgt.effects.some(function (f) { return f.kind === 'slow' || f.kind === 'stun'; }) ? 0.4 : 0);
    if (crit) amt *= (src.critMul || 1.5);
    if (crit && src.pet && src.frenzy) addEffect(src, { kind: 'haste', amt: 0.30, turns: 2, src: src.id, name: 'Неистовство', icon: '🐺' });
    if (src.effects && src.effects.some(function (f) { return f.kind === 'bwrath'; })) amt *= 1.5;
    if (src.effects && src.effects.some(function (f) { return f.kind === 'deathwish'; })) amt *= DEATHWISH.dealt;
    if (src.effects && src.effects.some(function (f) { return f.kind === 'arcpow'; })) amt *= 1.3;
    if (src.pomNow && !opt.auto && !opt.dot) amt *= 2;
    if (src.effects && src.effects.some(function (f) { return f.kind === 'pi'; })) amt *= 1.2;
    if (src.side === 'enemy' && src.effects.some(function (f) { return f.kind === 'weak'; })) amt *= 0.85;
    if (type === 'magic') { const vl = tgt.effects.find(function (f) { return f.kind === 'vuln'; }); if (vl) amt *= 1 + vl.amt * vl.stacks; }
    if (tgt.effects.some(function (f) { return f.kind === 'deathwish'; })) amt *= DEATHWISH.taken;
    if (src.effects && src.effects.some(function (f) { return f.kind === 'howl'; })) amt *= 1.08;
    if (src.shadowform) amt *= 1.15;
    if (src.mdMagic && type === 'magic') amt *= 1 + src.mdMagic;
    if (src.raceAll) amt *= 1 + src.raceAll;
    if (src.mdAll) amt *= 1 + src.mdAll;
    const vgn = src.effects && src.effects.find(function (f) { return f.kind === 'veng'; }); if (vgn) amt *= 1 + 0.05 * vgn.stacks;
    if (tgt.mdTaken) amt *= 1 - tgt.mdTaken;
    if (src.effects && src.effects.some(function (f) { return f.kind === 'feldom'; })) amt *= 1.25;
    if (tgt.shadowform) amt *= 0.85;
    if (!opt.pure) amt *= 1 - (type === 'magic' ? tgt.res : (tgt.armor + (tgt.effects.some(function (f) { return f.kind === 'hot' && f.name === 'Демонический доспех'; }) ? 0.15 : 0) + tgt.effects.reduce(function (a, f) { return a + (f.kind === 'armorup' ? f.amt : 0); }, 0)) * armorMod(tgt));
    if (tgt.side === 'hero') amt *= 1 - (st.aura.dmgTaken || 0);
    if (src.side === 'hero' && tgt.side === 'enemy') amt *= heroDmgMod(st.encLevel - src.level);
    if (src.side === 'enemy' && tgt.side === 'hero') amt *= foeDmgMod(st.encLevel - tgt.level);
    tgt.effects.forEach(function (f) { if (f.kind === 'wall') amt *= f.mult; });
    amt = Math.max(1, Math.round(amt));
    let absorbed = 0;
    const gs = tgt.side === 'hero' ? tgt.effects.find(function (f) { return f.kind === 'healup' && f.cheat; }) : null;
    if (gs && tgt.hp - amt <= 0) {
      tgt.effects = tgt.effects.filter(function (f) { return f !== gs; });
      absorbed = amt; amt = 0;
      const back = Math.max(0, Math.round(tgt.maxHp * 0.5) - tgt.hp); tgt.hp += back;
      const gsrc = byId(st, gs.src); if (gsrc && gsrc.side === 'hero') gsrc.stats.heal += back;
      emit(st, { t: 'heal', src: gs.src, tgt: tgt.id, amt: back, crit: false, hot: false });
      log(st, tgt.name + ' спасён Духом-хранителем', 'skill');
    }
    const sh = tgt.effects.find(function (f) { return f.kind === 'shield'; });
    if (sh) {
      absorbed = Math.min(sh.amt, amt); sh.amt -= absorbed; amt -= absorbed;
      if (sh.amt <= 0) tgt.effects = tgt.effects.filter(function (f) { return f !== sh; });
    }
    const dealt = Math.min(tgt.hp, amt);
    tgt.hp -= dealt;
    if (dealt > 0 && !opt.dot && !opt.refl) {
      if (src.side === 'hero' && tgt.side === 'enemy' && opt.auto && !opt.seal && type === 'phys') gainRage(src, dealt * (src.cls === 'druid' && src.spec === 'bear' ? RAGE_CFG.bearAutoFactor : RAGE_CFG.autoFactor) / rageConversion(src.level), 'auto');
      if (src.side === 'enemy' && tgt.side === 'hero') gainRage(tgt, dealt * RAGE_CFG.takenFactor / rageConversion(tgt.level), 'taken');
    }
    if (src.side === 'hero') { src.stats.dmg += dealt; addThreat(tgt, src, (dealt + absorbed) * src.threatMult * (opt.threatBonus || 1) * (src.effects.some(function (f) { return f.kind === 'rf'; }) ? 1.6 : 1)); }
    if (tgt.side === 'hero') tgt.stats.taken += dealt;
    emit(st, { t: 'dmg', src: src.id, tgt: tgt.id, amt: dealt, absorbed, crit, dot: !!opt.dot, auto: !!opt.auto });
    if (st.test && src.side === 'hero' && !src.dummy) { const so = src.pet ? byId(st, src.owner) : src; st.series.push([so ? (so.stats.turns || 0) : 0, so ? so.id : src.id, dealt, crit ? 1 : 0]); }
    if (crit && !opt.dot && !opt.refl) { if (src.side === 'hero' && !src.pet && src.pids) onCrit(st, src, tgt, type); if (src.side === 'enemy' && tgt.side === 'hero' && tgt.pids && tgt.pids.indexOf('reckon') !== -1) tgt.extraAuto = true; }
    if (src.side === 'hero' && !opt.dot && !opt.refl && dealt > 0 && src.effects.some(function (f) { return f.kind === 'vemb'; })) alive(st.heroes).forEach(function (x) { heal(st, src, x, dealt * 0.10, { noCrit: true }); });
    if (tgt.side === 'hero' && src.side === 'enemy' && !opt.dot && !opt.refl && src.hp > 0) { const rfl = tgt.effects.find(function (f) { return f.kind === 'reflect'; }); if (rfl && tgt.hp > 0) damage(st, tgt, src, rfl.amt, { type: 'magic', refl: true, noCrit: true }); }
    if (src.side === 'hero' && !opt.dot && !opt.seal && !opt.refl && tgt.hp > 0 && src.effects.some(function (f) { return f.kind === 'seal'; }) && st.rnd() < 0.3) damage(st, src, tgt, 0.7, { type: 'magic', seal: true, noCrit: true });
    if (src.side === 'hero' && !opt.dot && src.hp > 0) { const bt = src.effects.find(function (f) { return f.kind === 'btheal'; }); if (bt) { heal(st, src, src, Math.round(src.maxHp * 0.02), { noCrit: true }); bt.charges--; if (bt.charges <= 0) src.effects = src.effects.filter(function (f) { return f !== bt; }); } }
    if (tgt.hp <= 0) kill(st, tgt);
    else if (src.cls === 'rogue' && tgt.side === 'enemy' && !opt.dot) breakArmor(src, tgt);
    return dealt + absorbed;
  }

  function heal(st, src, tgt, raw, opt) {
    opt = opt || {};
    if (tgt.hp <= 0) return 0;
    let amt = raw * (0.9 + st.rnd() * 0.2);
    const fav = src.effects && !opt.hot ? src.effects.find(function (f) { return f.kind === 'favor'; }) : null;
    const crit = fav ? true : (!opt.noCrit && st.rnd() < (src.crit || 0));
    if (fav) src.effects = src.effects.filter(function (f) { return f !== fav; });
    if (crit) amt *= 1.5;
    if (src.effects && src.effects.some(function (f) { return f.kind === 'pi'; })) amt *= 1.2;
    if (src.raceAll) amt *= 1 + src.raceAll;
    if (crit && src.pids) { if (src.pids.indexOf('illum') !== -1 && src.maxMana) src.mana = Math.min(src.maxMana, src.mana + 5); if (src.pids.indexOf('inspir') !== -1) addEffect(tgt, { kind: 'armorup', amt: 0.20, turns: 2, src: src.id, name: 'Вдохновение', icon: '✨' }); }
    const hr = tgt.effects.reduce(function (a, f) { return f.kind === 'healred' ? a + f.amt * (f.stacks || 1) : a; }, 0);
    if (hr) amt *= Math.max(0.1, 1 - hr);
    const hu = tgt.effects.reduce(function (a, f) { return f.kind === 'healup' ? a + f.amt : a; }, 0);
    if (hu) amt *= 1 + hu;
    amt = Math.round(amt);
    const eff = Math.min(amt, tgt.maxHp - tgt.hp);
    tgt.hp += eff;
    if (st.test && src.side === 'hero' && !src.dummy) st.hlog.push([src.stats.turns || 0, src.id, eff, amt - eff]);
    if (src.side === 'hero' && eff > 0) {
      src.stats.heal += eff;
      const foes = alive(st.enemies);
      foes.forEach(function (e) { addThreat(e, src, eff * 0.5 / Math.max(1, foes.length)); });
    }
    emit(st, { t: 'heal', src: src.id, tgt: tgt.id, amt: eff, crit, hot: !!opt.hot });
    return eff;
  }

  function kill(st, u) {
    u.hp = 0; u.effects = [];
    emit(st, { t: 'death', id: u.id });
    log(st, u.name + ' погибает', u.side === 'hero' ? 'bad' : 'good');
    if (u.side === 'enemy' && u.onDeath && !u.flags.spawned) {
      u.flags.spawned = true;
      const ne = spawnEnemy(st, u.onDeath);
      log(st, 'Из Крошилки выбирается ' + ne.name, 'yell');
      emit(st, { t: 'yell', src: ne.id });
    }
    if (u.side === 'hero') st.enemies.forEach(function (e) { delete e.threat[u.id]; });
  }

  function addEffect(u, eff) {
    u.effects = u.effects.filter(function (f) { return !(f.kind === eff.kind && f.src === eff.src && f.name === eff.name); });
    u.effects.push(eff);
  }

  /* Неистовство: скорость растёт ступенями по 10% потерянного ХП, максимум +60% при ХП ниже 10%; лечение возвращает скорость. */
  const FURY = { step: 0.0667, max: 9 };
  const DEATHWISH = { dealt: 1.20, taken: 1.05 };
  function effSpd(u) {
    let m = 1;
    if (u.fury) m += Math.min(FURY.max, Math.max(0, Math.floor((1 - u.hp / u.maxHp) * 10))) * FURY.step;
    const hs = u.effects ? u.effects.find(function (f) { return f.kind === 'haste'; }) : null;
    if (hs) m += hs.amt;
    const sl = u.effects ? u.effects.find(function (f) { return f.kind === 'slow'; }) : null;
    if (sl) m -= sl.amt;
    const ab = u.effects ? u.effects.find(function (f) { return f.kind === 'arcb'; }) : null;
    if (ab) m -= 0.05 * ab.stacks;
    return u.spd * m;
  }
  function nextActor(st) {
    const units = all(st).filter(function (u) { return u.hp > 0; });
    let ready = units.filter(function (u) { return u.tm >= TM; });
    if (!ready.length) {
      let ticks = Infinity;
      units.forEach(function (u) { ticks = Math.min(ticks, Math.ceil((TM - u.tm) / effSpd(u))); });
      units.forEach(function (u) { u.tm += effSpd(u) * ticks; });
      ready = units.filter(function (u) { return u.tm >= TM; });
    }
    ready.sort(function (a, b) { return b.tm - a.tm; });
    return ready[0];
  }

  function turnOrder(st, n) {
    const sim = all(st).filter(function (u) { return u.hp > 0; }).map(function (u) { return { id: u.id, tm: u.tm, spd: effSpd(u) }; });
    const out = [];
    for (let i = 0; i < n && sim.length; i++) {
      let ready = sim.filter(function (u) { return u.tm >= TM; });
      if (!ready.length) {
        let ticks = Infinity;
        sim.forEach(function (u) { ticks = Math.min(ticks, Math.ceil((TM - u.tm) / u.spd)); });
        sim.forEach(function (u) { u.tm += u.spd * ticks; });
        ready = sim.filter(function (u) { return u.tm >= TM; });
      }
      ready.sort(function (a, b) { return b.tm - a.tm; });
      ready[0].tm -= TM;
      out.push(ready[0].id);
    }
    return out;
  }

  function startTurn(st, u) {
    if (u.maxEnergy) u.energy = Math.min(u.maxEnergy, Math.max(0, Number.isFinite(u.energy) ? u.energy : 0) + ENERGY_CFG.regen);
    if (u.maxRage) u.rageTakenThisTurn = 0;
    Object.keys(u.cd).forEach(function (k) { if (u.cd[k] > 0) u.cd[k]--; });
    if (u.side === 'hero' && u.maxMana) {
      u.mana = Math.min(u.maxMana, u.mana + MANA_CFG.regen);
      if (u.oom && u.mana >= u.maxMana * MANA_CFG.oomBack) u.oom = false;
    }
    const list = u.effects.slice();
    for (let i = 0; i < list.length; i++) {
      const f = list[i];
      if (u.hp <= 0) break;
      if (u.effects.indexOf(f) === -1) continue;
      const src = f.src ? byId(st, f.src) : null;
      if (f.kind === 'dot' && src) { const mm = f.ramp ? f.ramp[Math.min(f.tick || 0, f.ramp.length - 1)] : f.mult; f.tick = (f.tick || 0) + 1; if (src.pids && src.pids.indexOf('nightfall') !== -1 && f.name === 'Порча' && st.rnd() < 0.10) addEffect(src, { kind: 'nfb', turns: 4, src: src.id, name: 'Сумерки', icon: '🌌' }); const dd = damage(st, src, u, mm, { type: f.bleed ? 'phys' : 'magic', dot: true, pure: !!f.bleed }); if (f.vt && dd) alive(st.heroes).forEach(function (x) { if (x.maxMana) x.mana = Math.min(x.maxMana, x.mana + dd * 0.05); }); }
      if (f.kind === 'hot' && src && src.hp > 0) heal(st, src, u, f.amt, { hot: true });
    }
    if (u.hp <= 0) return;
    u.effects = u.effects.filter(function (f) { if (f.kind === 'enrage') return true; f.turns--; return f.turns > 0; });
  }

  /* Автоатака: базовый удар героя перед умением, вне слотов умений, нажимать не нужно.
     У ближнего боя удар оружием, у дальнего жезлом (ванд) или луком. Числа ориентировочные, баланс потом. */
  const AUTO_MULT = 0.2;
  /* Временная компенсация общего урона после урезания автоатак: атака всех героев ×1,35 (снимется при общем проходе баланса). */
  const ATKMUL = 1.35;
  const AUTO_DEFAULT = { name: 'Удар оружием', icon: '🗡️', type: 'phys' };
  const AUTO = {
    warrior: { name: 'Удар оружием', icon: '🗡️', type: 'phys' },
    paladin: { name: 'Удар оружием', icon: '🔨', type: 'phys' },
    rogue: { name: 'Удар кинжалом', icon: '🔪', type: 'phys' },
    druid: { name: 'Удар посохом', icon: '🪵', type: 'phys' },
    hunter: { name: 'Выстрел из лука', icon: '🏹', type: 'phys', ranged: true },
    mage: { name: 'Удар жезлом', icon: '🪄', type: 'magic', ranged: true },
    priest: { name: 'Удар жезлом', icon: '🪄', type: 'magic', ranged: true },
    warlock: { name: 'Удар жезлом', icon: '🪄', type: 'magic', ranged: true },
  };
  function autoOf(cls, spec) { const c = CLASSES[cls], sp = c && c.specs[spec]; return (sp && sp.auto) || AUTO[cls] || AUTO_DEFAULT; }
  const HEALAUTO = { frac: 0.55, below: 0.7, mana: 3 };
  function autoAttack(st, h) {
    if (h.hp <= 0 || !alive(st.enemies).length) return;
    if (h.role === 'heal' && (!h.maxMana || h.mana >= HEALAUTO.mana)) {
      const lw = lowestAlly(st);
      if (lw && lw.hp / lw.maxHp < HEALAUTO.below) {
        if (h.maxMana) h.mana -= HEALAUTO.mana;
        emit(st, { t: 'auto', src: h.id, name: h.cls === 'paladin' ? 'Вспышка Света' : 'Малое исцеление', icon: h.cls === 'paladin' ? '✨' : '💧' });
        heal(st, h, lw, h.healPow * HEALAUTO.frac);
        return;
      }
    }
    const a = autoOf(h.cls, h.spec), tank = h.role === 'tank', n = a.hits || 1;
    let t = tank ? tankTarget(st, h) : focusEnemy(st);
    emit(st, { t: 'auto', src: h.id, name: a.name, icon: a.icon });
    for (let i = 0; i < n; i++) {
      if (t.hp <= 0) { if (!alive(st.enemies).length) break; t = tank ? tankTarget(st, h) : focusEnemy(st); }
      damage(st, h, t, AUTO_MULT / n * (h.autoMul || 1), { type: a.type, auto: true, threatBonus: tank ? 1.3 : 1 });
    }
  }

  function use(st, h, skill) {
    if (skill === 'revenge' && !(h.revengeReady > 0)) return false;
    if (!canPayRage(h, skill)) return false;
    if (!canPayEnergy(h, skill)) return false;
    if (Object.prototype.hasOwnProperty.call(ENERGY_COST, skill)) { h.energy = Math.max(0, h.energy - ENERGY_COST[skill]); h.totalEnergySpent = (h.totalEnergySpent || 0) + ENERGY_COST[skill]; }
    if (Object.prototype.hasOwnProperty.call(RAGE_COST, skill)) {
      const paid = rageCost(h, skill);
      h.rage = Math.max(0, h.rage - paid);
      h.lastRageSpent = paid;
      if (['shieldslam','thunder','bloodthirst','whirlwind','maul','swipe'].indexOf(skill) !== -1) h.lastMainRageSkill = skill;
      h.totalRageSpent = (h.totalRageSpent || 0) + paid;
    }
    if (skill === 'revenge') h.revengeReady = 0;
    h.cd[skill] = SKILLS[skill].cd;
    if (h.arcConc && st.rnd() < 0.10) { const ks = Object.keys(h.cd).filter(function (k) { return k !== skill && h.cd[k] > 0; }); if (ks.length) { const k = ks[Math.floor(st.rnd() * ks.length)]; h.cd[k] = 0; log(st, h.name + ': «Чародейская концентрация» сбросила перезарядку «' + SKILLS[k].name + '»', 'good'); } }
    if (skill !== 'pom') { const pm = h.effects.find(function (f) { return f.kind === 'pom'; }); if (pm) { h.effects = h.effects.filter(function (f) { return f !== pm; }); h.pomNow = true; } }
    const c = SKILLS[skill].mana;
    if (c && h.maxMana) h.mana = Math.max(0, h.mana - c);
    emit(st, { t: 'skill', src: h.id, skill });
    return true;
  }

  function focusEnemy(st) { const f = alive(st.enemies); f.sort(function (a, b) { return a.hp - b.hp; }); return f[0]; }
  function tankTarget(st, h) { const f = alive(st.enemies); return f.find(function (e) { return e.aggro !== h.id; }) || focusEnemy(st); }
  function lowestAlly(st) { const a = alive(st.heroes); a.sort(function (x, y) { return pct(x) - pct(y); }); return a[0]; }
  function mostTargeted(st) {
    const counts = {};
    alive(st.enemies).forEach(function (e) { if (e.aggro) counts[e.aggro] = (counts[e.aggro] || 0) + (e.boss ? 3 : 1); });
    let best = null, bestV = 0;
    alive(st.heroes).forEach(function (h) { if ((counts[h.id] || 0) > bestV) { bestV = counts[h.id]; best = h; } });
    return best;
  }

  function doTaunt(st, h, skill, one) {
    const s = SKILLS[skill];
    (one ? [one] : alive(st.enemies)).forEach(function (e) {
      let max = 0;
      Object.keys(e.threat).forEach(function (k) { max = Math.max(max, e.threat[k]); });
      e.threat[h.id] = max * 1.1 + 50;
      e.effects = e.effects.filter(function (f) { return f.kind !== 'taunt'; });
      e.effects.push({ kind: 'taunt', target: h.id, turns: 3, name: s.name + ': ' + h.name, icon: s.icon });
    });
    log(st, h.name + ': «' + s.name + '», все враги переключаются на него', 'skill');
  }

  function aoe(st, h, mult, opt) { alive(st.enemies).forEach(function (e) { damage(st, h, e, mult, opt); }); }
  function dot(st, h, t, skill, mult) { const s = SKILLS[skill]; addEffect(t, { kind: 'dot', mult, turns: 3, src: h.id, name: s.name, icon: s.icon }); }
  function hot(st, h, t, skill, amt, turns) { const s = SKILLS[skill]; addEffect(t, { kind: 'hot', amt, turns: turns || 3, src: h.id, name: s.name, icon: s.icon }); }
  function mitig(h, skill, mult, turns) { const s = SKILLS[skill]; addEffect(h, { kind: 'wall', mult, turns, src: h.id, name: s.name, icon: s.icon }); }

  function heroAct(st, h) {
    const foes = alive(st.enemies);
    if (!foes.length) return;
    const multi = foes.length >= 2;
    const low = lowestAlly(st);
    const t = focusEnemy(st);
    const sk = function (s) {
      if (h.skills.indexOf(s) === -1 || h.cd[s]) return false;
      const c = SKILLS[s].mana;
      if (c && h.maxMana && h.mana < c) { h.oomHit = true; return false; }
      return canPayRage(h, s) && canPayEnergy(h, s);
    };

    switch (h.ai) {
      case 'warrior_prot': {
        const looseE = foes.filter(function(e){return e.aggro && e.aggro !== h.id;}).sort(function(a,b){return b.atk-a.atk;})[0];
        if (sk('wall') && pct(h)<0.4) { if(!use(st,h,'wall'))return;mitig(h,'wall',0.5,3);return; }
        if (sk('taunt') && looseE) { if(!use(st,h,'taunt'))return;doTaunt(st,h,'taunt',looseE);return; }
        if (multi && h.lastMainRageSkill === 'thunder' && sk('shieldslam')) { if(!use(st,h,'shieldslam'))return;damage(st,h,tankTarget(st,h),1.4,{threatBonus:2});return; }
        if (sk('thunder') && multi) { if(!use(st,h,'thunder'))return;aoe(st,h,0.55,{threatBonus:1.5});alive(st.enemies).forEach(function(e){addEffect(e,{kind:'slow',amt:0.10,turns:3,src:h.id,name:'Удар грома',icon:'⚡'});});return; }
        if (sk('shieldslam')) { if(!use(st,h,'shieldslam'))return;damage(st,h,tankTarget(st,h),1.4,{threatBonus:2});return; }

  function enemyTarget(st, e) {
    refreshAggro(st, e);
    let t = e.aggro ? byId(st, e.aggro) : null;
    if (!t || t.hp <= 0) {

      t = a[Math.floor(st.rnd() * a.length)];
      if (t) { e.threat[t.id] = (e.threat[t.id] || 0) + 1; e.aggro = t.id; }
    }
    return t;
  }

  function enrageIn(st) {
    if (!st.enrage) return null;
    if (st.actions < st.enrage.after) return st.enrage.after - st.actions;
    return st.enrage.every - ((st.actions - st.enrage.after) % st.enrage.every);
  }
  function enrageTick(st) {
    const c = st.enrage; if (!c) return;
    if (st.actions >= c.after && (st.actions - c.after) % c.every === 0) {
      st.enrageStacks++;
      const pct = Math.round(st.enrageStacks * c.step * 100);
      log(st, st.enrageStacks === 1 ? 'Бой затянулся: враги приходят в ярость, их урон растёт каждые ' + c.every + ' ходов (сейчас +' + pct + '%)' : 'Ярость врагов растёт: урон +' + pct + '%', 'yell');
      emit(st, { t: 'enrage', stacks: st.enrageStacks, pct: pct });
    }
  }

  function step(st) {
    st.events = [];
    if (st.over) return st.events;
    const u = nextActor(st);
    if (!u) return st.events;
    u.tm -= TM;
    st.actions++;
    st.current = u.id;
    emit(st, { t: 'turn', id: u.id });
    enrageTick(st);
    startTurn(st, u);
    if (u.side === 'hero' && !u.pet) petsTick(st, u);
    if (u.hp > 0 && !checkEnd(st)) { emit(st, { t: 'act', id: u.id }); if (st.test && u.side === 'hero' && !u.pet && !u.dummy) { u.stats.turns = (u.stats.turns || 0) + 1; if (u.isMain || (u === st.heroes[0] && !st.heroes.some(function (x) { return x.isMain; }))) st.mainTurns++; }
    if (u.pet) petAct(st, u); else if (u.dummy) { emit(st, { t: 'idle', id: u.id }); } else if (u.side === 'hero') { racialAct(st, u); if (u.effects.some(function (f) { return f.kind === 'stun'; })) { emit(st, { t: 'eskill', src: u.id, id: 'stun', name: 'Оглушён', icon: '💫' }); log(st, u.name + ' оглушён и пропускает ход', 'info'); } else if (u.role !== 'heal' && !autoOf(u.cls, u.spec).ranged && u.effects.some(function (f) { return f.kind === 'root'; })) { emit(st, { t: 'eskill', src: u.id, id: 'root', name: 'Обездвижен', icon: '🕸️' }); log(st, u.name + ' обездвижен и не может подойти к врагу', 'info'); } else { if (!u.effects.some(function (f) { return f.kind === 'hidden'; })) { autoAttack(st, u); if (u.extraAuto) { u.extraAuto = false; autoAttack(st, u); } } heroAct(st, u); u.pomNow = false; } } else if (u.effects.some(function (f) { return f.kind === 'stun'; })) { emit(st, { t: 'eskill', src: u.id, id: 'stun', name: 'Оглушён', icon: '💫' }); log(st, u.name + ' оглушён и пропускает ход', 'info'); } else enemyAct(st, u); }
    if (u.revengeReady > 0) u.revengeReady--;
    if (u.side === 'hero' && u.oomHit) {
      u.oomHit = false;
      if (!u.oom) { u.oom = true; emit(st, { t: 'oom', id: u.id }); log(st, u.name + ': не хватает маны для одного из умений', 'bad'); }
    }
    (st.pets || []).forEach(function (p) { const o = byId(st, p.owner); if (o && o.hp <= 0 && !p.gone) { p.hp = 0; p.gone = true; } });
    st.enemies.forEach(function (e) { if (e.hp > 0) refreshAggro(st, e); });
    checkEnd(st);
    if (!st.over && st.actions >= MAX_ACTIONS) { st.over = true; st.result = 'lose'; log(st, 'Бой затянулся. Отряд отступает', 'bad'); }
    return st.events;
  }

  function checkEnd(st) {
    if (st.over) return true;
    if (st.test) {
      if (!alive(st.heroes).filter(function (h) { return !h.dummy; }).length) { st.over = true; st.result = 'lose'; log(st, 'Отряд погиб', 'bad'); return true; }
      if (st.test.limit && st.mainTurns >= st.test.limit) { st.over = true; st.result = 'test'; log(st, 'Проба окончена', 'good'); return true; }
      return false;
    }
    if (!alive(st.enemies).length) { st.over = true; st.result = 'win'; log(st, 'Победа!', 'good'); return true; }
    if (!alive(st.heroes).length) { st.over = true; st.result = 'lose'; log(st, 'Отряд погиб', 'bad'); return true; }
    return false;
  }

  function dungeonXp(level, key) { return DUNGEONS[key].rooms.reduce(function (a, r) { return a + xpFor(level, ENCOUNTERS[r]); }, 0); }
  function newRun(dk, party, main, level, seed) {
    const D = DUNGEONS[dk];
    return { dk: dk, D: D, party: party, main: main, level: level, rooms: D.rooms.slice(), idx: 0, rests: D.rests, restsMax: D.rests, carry: null, state: 'room', loot: [], xp: 0, kills: 0, seed: seed || 1, rnd: makeRng((seed || 1) + 99), usedRest: 0 };
  }
  function roomKey(run) { return run.rooms[run.idx]; }
  function startRoom(run, seed) {
    return createBattle({ party: run.party, main: run.main, encounter: roomKey(run), level: run.level, seed: seed || ((run.seed + run.idx * 7919) >>> 0), carry: run.carry });
  }
  function endRoom(run, st) {
    const key = roomKey(run);
    if (st.result === 'win') {
      run.kills++;
      run.xp += xpFor(run.level, ENCOUNTERS[key]);
      const table = LOOT[key] || [];
      if (table.length) run.loot.push({ room: key, item: table[Math.floor(run.rnd() * table.length)] });
      const rc = RECOVERY;
      run.carry = st.heroes.map(function (h) {
        if (h.hp <= 0) return { hp: 0, maxHp: h.maxHp, mana: h.mana, maxMana: h.maxMana };
        return { hp: Math.min(h.maxHp, h.hp + h.maxHp * rc.hp), maxHp: h.maxHp, mana: h.maxMana ? Math.min(h.maxMana, h.mana + h.maxMana * rc.mana) : 0, maxMana: h.maxMana };
      });
      run.idx++;
      run.state = run.idx >= run.rooms.length ? 'cleared' : 'between';
    } else {
      run.state = 'wipe';
    }
  }
  function runStatus(run) {
    const c = run.carry || [];
    if (!c.length) return { avgHp: 1, dead: 0, minMana: null };
    const avgHp = c.reduce(function (a, x) { return a + x.hp / x.maxHp; }, 0) / c.length;
    const mana = c.filter(function (x) { return x.maxMana > 0; }).map(function (x) { return x.mana / x.maxMana; });
    return { avgHp: avgHp, dead: c.filter(function (x) { return x.hp <= 0; }).length, minMana: mana.length ? Math.min.apply(null, mana) : null };
  }
  function restRun(run) {
    if (run.rests <= 0 || !run.carry) return false;
    run.rests--; run.usedRest++;
    run.carry = run.carry.map(function (x) { return { hp: x.maxHp, maxHp: x.maxHp, mana: x.maxMana, maxMana: x.maxMana }; });
    return true;
  }
  function autoDecide(run) {
    const s = runStatus(run);
    const weak = s.avgHp < RESTAT.hp || s.dead > 0 || (s.minMana !== null && s.minMana < RESTAT.mana);
    if (!weak) return 'next';
    if (run.rests > 0) return 'rest';
    return (s.avgHp < 0.35 || s.dead >= 2) ? 'exit' : 'next';
  }
  function simulateRun(run) {
    const log = [];
    let guard = 0;
    while ((run.state === 'room' || run.state === 'between') && guard++ < 40) {
      if (run.state === 'between') {
        const d = autoDecide(run);
        log.push(d);
        if (d === 'rest') restRun(run);
        else if (d === 'exit') { run.state = 'exit'; break; }
        run.state = 'room';
        continue;
      }
      const st = startRoom(run);
      while (!st.over) step(st);
      endRoom(run, st);
    }
    return { state: run.state, kills: run.kills, rooms: run.rooms.length, xp: run.xp, loot: run.loot.slice(), rests: run.usedRest, decisions: log };
  }
